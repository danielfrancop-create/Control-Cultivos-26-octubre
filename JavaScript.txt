    <script>
        // --- INDEXEDDB & OFFLINE PWA MANAGER ---
        const DB_NAME = 'FincaSaucesOfflineDB';
        const DB_VERSION = 1;
        const STORE_NAME = 'pending_records';

        function openOfflineDB() {
            return new Promise((resolve, reject) => {
                if (!window.indexedDB) {
                    resolve(null);
                    return;
                }
                const request = indexedDB.open(DB_NAME, DB_VERSION);
                request.onupgradeneeded = (e) => {
                    const db = e.target.result;
                    if (!db.objectStoreNames.contains(STORE_NAME)) {
                        db.createObjectStore(STORE_NAME, { keyPath: 'id', autoIncrement: true });
                    }
                };
                request.onsuccess = (e) => resolve(e.target.result);
                request.onerror = (e) => reject(e.target.error);
            });
        }

        async function saveOfflineRecord(tipo, record) {
            try {
                const db = await openOfflineDB();
                if (!db) return null;
                return new Promise((resolve, reject) => {
                    const tx = db.transaction(STORE_NAME, 'readwrite');
                    const store = tx.objectStore(STORE_NAME);
                    const item = {
                        tipo: tipo,
                        record: record,
                        estado: 'pendiente_sincronizacion',
                        timestamp: new Date().getTime()
                    };
                    const req = store.add(item);
                    req.onsuccess = (e) => {
                        updatePendingSyncBadge();
                        resolve(e.target.result);
                    };
                    req.onerror = (e) => reject(e.target.error);
                });
            } catch (err) {
                console.error("Error al guardar en IndexedDB:", err);
                return null;
            }
        }

        async function getOfflineRecords() {
            try {
                const db = await openOfflineDB();
                if (!db) return [];
                return new Promise((resolve, reject) => {
                    const tx = db.transaction(STORE_NAME, 'readonly');
                    const store = tx.objectStore(STORE_NAME);
                    const req = store.getAll();
                    req.onsuccess = (e) => resolve(e.target.result || []);
                    req.onerror = (e) => reject(e.target.error);
                });
            } catch (err) {
                console.error("Error leyendo de IndexedDB:", err);
                return [];
            }
        }

        async function deleteOfflineRecord(id) {
            try {
                const db = await openOfflineDB();
                if (!db) return;
                return new Promise((resolve, reject) => {
                    const tx = db.transaction(STORE_NAME, 'readwrite');
                    const store = tx.objectStore(STORE_NAME);
                    const req = store.delete(id);
                    req.onsuccess = () => {
                        updatePendingSyncBadge();
                        resolve();
                    };
                    req.onerror = (e) => reject(e.target.error);
                });
            } catch (err) {
                console.error("Error borrando registro de IndexedDB:", err);
            }
        }

        async function updatePendingSyncBadge() {
            const records = await getOfflineRecords();
            const badge = document.getElementById("pendingSyncBadge");
            const btn = document.getElementById("syncOfflineBtn");
            const count = records.length;
            if (badge) badge.textContent = count;
            if (btn) {
                if (count > 0) {
                    btn.classList.remove("hidden");
                } else {
                    btn.classList.add("hidden");
                }
            }
        }

            if (event) event.stopPropagation();
            const menu = document.getElementById("desktopNavDropdownMenu");
            if (menu) {
                menu.classList.toggle("hidden");
            }
        }

        document.addEventListener("click", function(event) {
            const dropdownBtn = document.getElementById("desktopNavDropdownBtn");
            const dropdownMenu = document.getElementById("desktopNavDropdownMenu");
            if (dropdownMenu && !dropdownMenu.classList.contains("hidden")) {
                if (dropdownBtn && !dropdownBtn.contains(event.target) && !dropdownMenu.contains(event.target)) {
                    dropdownMenu.classList.add("hidden");
                }
            }
        });

        async function syncOfflineData() {
            if (!navigator.onLine) {
                alert("Sin conexión a Internet. Conéctate a una red para sincronizar con Google Sheets.");
                return;
            }

            const pending = await getOfflineRecords();
            if (pending.length === 0) {
                alert("No hay registros pendientes por sincronizar.");
                return;
            }

            showLoader(true, `Sincronizando ${pending.length} registro(s) pendiente(s)...`);

            let syncedCount = 0;
            for (const item of pending) {
                try {
                    await new Promise((resolve, reject) => {
                        if (typeof google !== 'undefined' && google.script && google.script.run) {
                            let runner = google.script.run
                                .withSuccessHandler((res) => {
                                    if (res && res.success) resolve(res);
                                    else reject(res ? res.error : "Error en servidor");
                                })
                                .withFailureHandler((err) => reject(err));

                            if (item.tipo === 'fotosLotes' || item.tipo === 'foto') {
                                runner.guardarFotoLote(item.record);
                            } else if (item.tipo === 'gastosPersonales') {
                                runner.guardarGastoPersonal(item.record);
                            } else {
                                runner.guardarRegistro(item.tipo, item.record);
                            }
                        } else {
                            setTimeout(() => resolve({ success: true }), 300);
                        }
                    });
                    await deleteOfflineRecord(item.id);
                    syncedCount++;
                } catch (err) {
                    console.error("Error sincronizando registro ID:", item.id, err);
                }
            }

            showLoader(false);
            await updatePendingSyncBadge();

            if (syncedCount > 0) {
                alert(`¡Sincronización completada! Se subieron ${syncedCount} registro(s) exitosamente a Google Sheets / Drive.`);
                fetchData();
            } else {
                alert("Ocurrió un problema al intentar sincronizar los registros.");
            }
        }

        window.addEventListener('online', () => {
            updatePendingSyncBadge();
            syncOfflineData();
        });
        window.addEventListener('offline', () => {
            updatePendingSyncBadge();
        });

        let rawData = null;

        let doughnutChartInstance = null;

        function formatCOP(val) {
            const num = Math.round(Number(val) || 0);
            const isNegative = num < 0;
            const absoluteValue = Math.abs(num);
            const formatted = absoluteValue.toLocaleString('es-CO', {
                minimumFractionDigits: 0,
                maximumFractionDigits: 0
            });
            return `${isNegative ? '-' : ''}$${formatted}`;
        }

        function updateSyncTimestamp(customDateStr) {
            let formattedTime = "";
            if (customDateStr) {
                formattedTime = customDateStr;
            } else {
                const now = new Date();
                const year = now.getFullYear();
                const month = String(now.getMonth() + 1).padStart(2, '0');
                const day = String(now.getDate()).padStart(2, '0');
                const hours = String(now.getHours()).padStart(2, '0');
                const minutes = String(now.getMinutes()).padStart(2, '0');
                formattedTime = `${year}-${month}-${day} ${hours}:${minutes}`;
            }

            const headerSubtitle = document.getElementById("headerSubtitle");
            if (headerSubtitle) {
                const currentText = headerSubtitle.textContent || "";
                let baseUbicacion = "Ubicación: Manizales, Caldas";
                if (currentText.includes("|")) {
                    baseUbicacion = currentText.split("|")[0].trim();
                }
                headerSubtitle.textContent = `${baseUbicacion} | Última Sincronización: ${formattedTime}`;
            }
        }

        function fetchData(forceRefresh = false) {
            // Check sessionStorage cache first for instant initial load if not forceRefresh
            const CACHE_KEY = "finca_dashboard_cache_data";
            if (!forceRefresh) {
                try {
                    const cachedStr = sessionStorage.getItem(CACHE_KEY);
                    if (cachedStr) {
                        const cachedData = JSON.parse(cachedStr);
                        if (cachedData && typeof cachedData === 'object') {
                            rawData = cachedData;
                            handleDataLoaded(rawData);
                            showLoader(false);
                            // Asynchronously fetch fresh data in background without blocking UI
                            if (typeof google !== 'undefined' && google.script && google.script.run) {
                                google.script.run
                                    .withSuccessHandler(function(response) {
                                        if (response && response.success) {
                                            rawData = response.data;
                                            try { sessionStorage.setItem(CACHE_KEY, JSON.stringify(rawData)); } catch(e){}
                                            updateConfigHeader();
                                            applyFilters();
                                        }
                                    })
                                    .getDashboardData();
                            }
                            return;
                        }
                    }
                } catch (e) {
                    console.warn("Error leyendo de sessionStorage:", e);
                }
            }

            // Visual spin animation on refresh icon
            const iconDesktop = document.getElementById("iconRefrescar");
            const iconMobile = document.getElementById("iconRefrescarMobile");
            if (iconDesktop) iconDesktop.classList.add("animate-spin");
            if (iconMobile) iconMobile.classList.add("animate-spin");

            showLoader(true, "Refrescando datos desde el servidor...");
            hideError();

            const stopSpin = () => {
                if (iconDesktop) iconDesktop.classList.remove("animate-spin");
                if (iconMobile) iconMobile.classList.remove("animate-spin");
            };

            if (typeof google === 'undefined' || !google.script || !google.script.run) {
                console.warn("Entorno local detectado. Cargando datos simulados...");
                setTimeout(() => {
                    const mock = getMockData();
                    handleDataLoaded(mock);
                    stopSpin();
                    updateSyncTimestamp();
                }, 800);
                return;
            }

            google.script.run
                .withSuccessHandler(function(response) {
                    stopSpin();
                    if (response && response.success) {
                        rawData = response.data;
                        try { sessionStorage.setItem(CACHE_KEY, JSON.stringify(rawData)); } catch(e){}
                        updateConfigHeader();
                        updateSyncTimestamp();
                        applyFilters();
                        fetchSecurityCredentialsBackend();
                        showLoader(false);
                    } else {
                        showError(response ? response.error : "No se obtuvo respuesta válida.");
                    }
                })
                .withFailureHandler(function(err) {
                    stopSpin();
                    showError(err.toString());
                })
                .getDashboardData();
        }

        // Global Lot Matching Helper
        function normalizeLoteStr(str) {
            if (!str) return "";
            let s = String(str).toUpperCase().trim();
            s = s.replace(/\(CONSOLIDADO.*?\)/g, "").trim();
            s = s.replace(/LOTE\s*/g, "").replace(/\-/g, "").replace(/\s+/g, "");
            return s;
        }

        function isLoteMatch(itemLoteStr, targetLoteStr) {
            const target = String(targetLoteStr || "Todos").trim();
            if (!target || target === "Todos") return true;

            const normTarget = normalizeLoteStr(target);
            const normItem = normalizeLoteStr(itemLoteStr);

            if (normTarget === "TODOS" || normTarget === "") return true;
            if (normItem === normTarget) return true;

            if (/^\d+$/.test(normTarget)) {
                const pattern = new RegExp("^" + normTarget + "[A-Z]+$");
                if (pattern.test(normItem)) return true;
            }

            return false;
        }

        // Helper to update the Ciclo de Cultivo dropdown when a Lot is selected
        function updateCicloDropdown(selectedLote) {
            const container = document.getElementById("filterCicloContainer");
            const cicloSelect = document.getElementById("filterCiclo");
            if (!container || !cicloSelect) return;

            if (!selectedLote || selectedLote === "Todos") {
                container.classList.add("hidden");
                cicloSelect.innerHTML = '<option value="Todos">Todos los Ciclos (Histórico General)</option>';
                cicloSelect.value = "Todos";
                return;
            }

            container.classList.remove("hidden");
            const previousVal = cicloSelect.value;
            const cycleCodesSet = new Set();

            if (rawData && rawData.ciclos) {
                (rawData.ciclos || []).forEach(c => {
                    if (isLoteMatch(c.Lote || c.lote || c.Lote_Principal || c.lote_principal, selectedLote)) {
                        const code = String(c.Codigo_General || c.codigo_general || c.Codigo_Inicial || c.codigo_inicial || "").trim().toUpperCase();
                        if (code) cycleCodesSet.add(code);
                    }
                });
            }

            // Also check tracking codes from initial expenses, development, and sales for this lote
            ['gastosIniciales', 'gastosDesarrollo', 'ventas'].forEach(key => {
                if (rawData && rawData[key]) {
                    (rawData[key] || []).forEach(item => {
                        if (isLoteMatch(item.Lote || item.lote, selectedLote)) {
                            const code = String(item.Codigo_Seguimiento || item.codigo_seguimiento || item.Codigo_General || item.codigo_general || "").trim().toUpperCase();
                            if (code && code !== "-" && code !== "SIN CÓDIGO") cycleCodesSet.add(code);
                        }
                    });
                }
            });

            let optionsHTML = '<option value="Todos">Todos los Ciclos (Histórico General)</option>';
            Array.from(cycleCodesSet).sort().forEach(code => {
                optionsHTML += `<option value="${code}">${code}</option>`;
            });

            cicloSelect.innerHTML = optionsHTML;
            if (previousVal && Array.from(cycleCodesSet).includes(previousVal)) {
                cicloSelect.value = previousVal;
            } else {
                cicloSelect.value = "Todos";
            }
        }

        // Selector synchronization helper (keeps header select and filter bar select aligned)
        function syncSelectors(source) {
            const headerSelect = document.getElementById("loteSelector");
            const barSelect = document.getElementById("filterLote");
            const cfSelect = document.getElementById("cashflowLoteSelector");

            let targetVal = "Todos";
            if (source === 'header' && headerSelect) targetVal = headerSelect.value;
            else if (source === 'bar' && barSelect) targetVal = barSelect.value;
            else if (source === 'cashflow' && cfSelect) targetVal = cfSelect.value;

            if (headerSelect && headerSelect.value !== targetVal) headerSelect.value = targetVal;
            if (barSelect && barSelect.value !== targetVal) barSelect.value = targetVal;
            if (cfSelect && cfSelect.value !== targetVal) cfSelect.value = targetVal;

            updateCicloDropdown(targetVal);
            applyFilters();
        }

        // Fast periods helper
        function setPeriod(period) {
            const today = new Date();
            const year = today.getFullYear();
            let desde = "";
            let hasta = "";

            if (period === 'este_mes') {
                const month = String(today.getMonth() + 1).padStart(2, '0');
                const lastDay = new Date(year, today.getMonth() + 1, 0).getDate();
                desde = `${year}-${month}-01`;
                hasta = `${year}-${month}-${String(lastDay).padStart(2, '0')}`;
            } else if (period === 'primer_semestre') {
                desde = `${year}-01-01`;
                hasta = `${year}-06-30`;
            } else if (period === 'segundo_semestre') {
                desde = `${year}-07-01`;
                hasta = `${year}-12-31`;
            } else if (period === 'todo') {
                desde = "";
                hasta = "";
                const filterLote = document.getElementById("filterLote");
                if (filterLote) filterLote.value = "Todos";
                const filterCultivo = document.getElementById("filterCultivo");
                if (filterCultivo) filterCultivo.value = "Todos";
                const filterCiclo = document.getElementById("filterCiclo");
                if (filterCiclo) filterCiclo.value = "Todos";
                syncSelectors('bar');
            }

            document.getElementById("filterFechaDesde").value = desde;
            document.getElementById("filterFechaHasta").value = hasta;
            applyFilters();
        }

        // Determine primary crop for any transaction
        function getCultivoForTx(item) {
            if (!item) return "N/A";
            
            // Check direct properties on transaction first
            const directCultivo = String(
                item.Cultivo_Semilla || item.cultivo_semilla || 
                item.Cultivo || item.cultivo || 
                item.Producto_Vendido || item.producto_vendido || ""
            ).trim();

            if (directCultivo) return directCultivo;

            if (!rawData) return "N/A";
            const lote = String(item.Lote || item.lote || "").trim();
            
            const trackingCode = String(item.Codigo_General || item.codigo_general || item.Codigo_Seguimiento || item.codigo_seguimiento || "").trim().toUpperCase();
            if (trackingCode && rawData.ciclos) {
                const cycle = (rawData.ciclos || []).find(c => 
                    String(c.Codigo_General || c.codigo_general || "").trim().toUpperCase() === trackingCode ||
                    String(c.Codigo_Inicial || c.codigo_inicial || "").trim().toUpperCase() === trackingCode ||
                    String(c.Codigo_Desarrollo || c.codigo_desarrollo || "").trim().toUpperCase() === trackingCode ||
                    String(c.Codigo_Ventas || c.codigo_ventas || "").trim().toUpperCase() === trackingCode
                );
                if (cycle) return String(cycle.Cultivo || cycle.cultivo || "").trim();
            }
            
            if (rawData.gastosIniciales) {
                const ini = (rawData.gastosIniciales || []).find(x => isLoteMatch(x.Lote || x.lote, lote));
                if (ini) return String(ini.Cultivo_Semilla || ini.cultivo_semilla || "").trim();
            }
            return "N/A";
        }

        function applyFilters() {
            if (!rawData) return;
            
            const selectedLote = document.getElementById("filterLote").value;
            const selectedCultivo = document.getElementById("filterCultivo").value;
            const selectedCiclo = document.getElementById("filterCiclo") ? document.getElementById("filterCiclo").value : "Todos";
            const fechaDesde = document.getElementById("filterFechaDesde").value;
            const fechaHasta = document.getElementById("filterFechaHasta").value;

            // Date validation helper
            const isDateInRange = (dateStr) => {
                if (!dateStr || dateStr === "-") return true;
                const d = dateStr.trim();
                if (fechaDesde && d < fechaDesde) return false;
                if (fechaHasta && d > fechaHasta) return false;
                return true;
            };

            // Lote selection helper using global isLoteMatch
            const checkLoteMatch = (loteStr) => {
                return isLoteMatch(loteStr, selectedLote);
            };

            // Ciclo filter helper
            const isCicloMatch = (item) => {
                if (!selectedCiclo || selectedCiclo === "Todos") return true;
                const code = String(item.Codigo_Seguimiento || item.codigo_seguimiento || item.Codigo_General || item.codigo_general || "").trim().toUpperCase();
                return code === selectedCiclo.trim().toUpperCase();
            };

            // Cultivo filter helper
            const isCultivoMatch = (item) => {
                if (selectedCultivo === "Todos") return true;
                const c = getCultivoForTx(item);
                return c.toLowerCase().indexOf(selectedCultivo.toLowerCase()) !== -1;
            };

            // Filter raw data
            const gInicialesFiltrados = (rawData.gastosIniciales || []).filter(item => {
                return checkLoteMatch(item.Lote || item.lote) && 
                       isCicloMatch(item) &&
                       isDateInRange(item.Fecha || item.fecha) && 
                       isCultivoMatch(item);
            });

            const gDesarrolloFiltrados = (rawData.gastosDesarrollo || []).filter(item => {
                return checkLoteMatch(item.Lote || item.lote) && 
                       isCicloMatch(item) &&
                       isDateInRange(item.Fecha || item.fecha) && 
                       isCultivoMatch(item);
            });

            const ventasFiltradas = (rawData.ventas || []).filter(item => {
                return checkLoteMatch(item.Lote || item.lote) && 
                       isCicloMatch(item) &&
                       isDateInRange(item.Fecha || item.fecha) && 
                       isCultivoMatch(item);
            });

            // Update Category Counter Cards UI
            const counterInicialesEl = document.getElementById("counterIniciales");
            if (counterInicialesEl) {
                counterInicialesEl.textContent = `${gInicialesFiltrados.length} registros (Total en el filtro)`;
            }

            const counterDesarrolloEl = document.getElementById("counterDesarrollo");
            if (counterDesarrolloEl) {
                counterDesarrolloEl.textContent = `${gDesarrolloFiltrados.length} registros (Total en el filtro)`;
            }

            const counterVentasEl = document.getElementById("counterVentas");
            if (counterVentasEl) {
                counterVentasEl.textContent = `${ventasFiltradas.length} registros (Total en el filtro)`;
            }

            // Perform calculations
            const totalGastosIniciales = gInicialesFiltrados.reduce((sum, item) => sum + (Number(item.Costo_Total_COP || item.costo_total_cop) || 0), 0);
            const totalGastosDesarrollo = gDesarrolloFiltrados.reduce((sum, item) => sum + (Number(item.Costo_Total_COP || item.costo_total_cop) || 0), 0);
            const totalVentas = ventasFiltradas.reduce((sum, item) => sum + (Number(item.Ingreso_Total_COP || item.ingreso_total_cop) || 0), 0);
            
            // Extract harvested units from ventes
            const totalUnidades = ventasFiltradas.reduce((sum, item) => sum + (Number(item.Unidades_Cosechadas || item.unidades_cosechadas) || 0), 0);
            
            const gastosTotales = totalGastosIniciales + totalGastosDesarrollo;
            const utilidadNeta = totalVentas - gastosTotales;
            
            // ROI calculation
            let roiPorcentaje = 0;
            if (gastosTotales > 0) {
                roiPorcentaje = (utilidadNeta / gastosTotales) * 100;
            }

            // Unit costs & margins
            const costoUnitario = totalUnidades > 0 ? (gastosTotales / totalUnidades) : 0;
            const precioVentaPromedio = totalUnidades > 0 ? (totalVentas / totalUnidades) : 0;
            const margenUnitario = precioVentaPromedio - costoUnitario;

            // Sum up total debts from bank loans
            const deudaTotalActiva = (rawData.prestamos || []).reduce((sum, item) => sum + (Number(item.Saldo_Por_Pagar || item.saldo_por_pagar) || 0), 0);

            // Update KPI Card UI elements
            // Tarjeta 1: Utilidad Neta Total
            const kpiUtilidadNetaEl = document.getElementById("kpiUtilidadNeta");
            if (kpiUtilidadNetaEl) kpiUtilidadNetaEl.textContent = formatCOP(utilidadNeta);
            
            const iconContainer = document.getElementById("kpiUtilidadIconContainer");
            if (iconContainer) {
                if (utilidadNeta >= 0) {
                    iconContainer.className = "p-2.5 bg-emerald-50 dark:bg-emerald-950 text-emerald-600 dark:text-emerald-400 rounded-xl";
                } else {
                    iconContainer.className = "p-2.5 bg-rose-50 dark:bg-rose-950 text-rose-600 dark:text-rose-400 rounded-xl";
                }
            }

            // Tarjeta 2: Costo Promedio por Unidad Cosechada
            const kpiCostoUnitarioEl = document.getElementById("kpiCostoUnitario");
            if (kpiCostoUnitarioEl) {
                kpiCostoUnitarioEl.textContent = totalUnidades > 0 ? `${formatCOP(costoUnitario)} / Unidad` : "$0 / Unidad";
            }
            const kpiCostoUnitarioSubtextEl = document.getElementById("kpiCostoUnitarioSubtext");
            if (kpiCostoUnitarioSubtextEl) {
                kpiCostoUnitarioSubtextEl.innerHTML = `Venta Prom: <span class="font-bold text-slate-800 dark:text-slate-100">${formatCOP(precioVentaPromedio)}</span> | Margen: <span class="font-bold ${margenUnitario >= 0 ?         function renderLotesTable(selectedLote) {
            const tableBody = document.getElementById("resumenLotesTableBody");
            if (!tableBody) return;
            tableBody.innerHTML = "";

            const selectedCultivo = document.getElementById("filterCultivo").value;
            const selectedCiclo = document.getElementById("filterCiclo") ? document.getElementById("filterCiclo").value : "Todos";
            const fechaDesde = document.getElementById("filterFechaDesde").value;
            const fechaHasta = document.getElementById("filterFechaHasta").value;

            // Date validation helper
            const isDateInRange = (dateStr) => {
                if (!dateStr || dateStr === "-") return true;
                const d = dateStr.trim();
                if (fechaDesde && d < fechaDesde) return false;
                if (fechaHasta && d > fechaHasta) return false;
                return true;
            };

            // Ciclo filter helper
            const isCicloMatch = (item) => {
                if (!selectedCiclo || selectedCiclo === "Todos") return true;
                const code = String(item.Codigo_Seguimiento || item.codigo_seguimiento || item.Codigo_General || item.codigo_general || "").trim().toUpperCase();
                return code === selectedCiclo.trim().toUpperCase();
            };

            // Cultivo filter helper
            const isCultivoMatch = (item) => {
                if (selectedCultivo === "Todos") return true;
                const c = getCultivoForTx(item);
                return c.toLowerCase().indexOf(selectedCultivo.toLowerCase()) !== -1;
            };

            let lotesAProcesar = [];
            if (selectedLote === "Todos") {
                for (let i = 1; i <= 16; i++) lotesAProcesar.push(`Lote ${i}`);
            } else {
                lotesAProcesar.push(selectedLote);
            }

            let renderedCount = 0;

            lotesAProcesar.forEach(lote => {
                // Find primary crop for this lot
                const iniRow = (rawData.gastosIniciales || []).find(item => isLoteMatch(item.Lote || item.lote, lote));
                const cultivo = iniRow ? (iniRow.Cultivo_Semilla || iniRow.cultivo_semilla || "Por Definir") : "Sin Registrar";

                // Filter transactions for this lote
                const gIniTotal = (rawData.gastosIniciales || [])
                    .filter(item => isLoteMatch(item.Lote || item.lote, lote) && isCicloMatch(item) && isDateInRange(item.Fecha || item.fecha) && isCultivoMatch(item))
                    .reduce((sum, item) => sum + (Number(item.Costo_Total_COP || item.costo_total_cop) || 0), 0);

                const gDesTotal = (rawData.gastosDesarrollo || [])
                    .filter(item => isLoteMatch(item.Lote || item.lote, lote) && isCicloMatch(item) && isDateInRange(item.Fecha || item.fecha) && isCultivoMatch(item))
                    .reduce((sum, item) => sum + (Number(item.Costo_Total_COP || item.costo_total_cop) || 0), 0);

                const costoTotal = gIniTotal + gDesTotal;
                
                const filteredVentas = (rawData.ventas || [])
                    .filter(item => isLoteMatch(item.Lote || item.lote, lote) && isCicloMatch(item) && isDateInRange(item.Fecha || item.fecha) && isCultivoMatch(item));

                const ventasTotal = filteredVentas.reduce((sum, item) => sum + (Number(item.Ingreso_Total_COP || item.ingreso_total_cop) || 0), 0);
                const unidadesTotal = filteredVentas.reduce((sum, item) => sum + (Number(item.Unidades_Cosechadas || item.unidades_cosechadas) || 0), 0);

                const ganancia = ventasTotal - costoTotal;
                
                let roi = 0;
                if (costoTotal > 0) {
                    roi = (ganancia / costoTotal) * 100;
                }

                const costoUnitario = unidadesTotal > 0 ? (costoTotal / unidadesTotal) : 0;
                const precioVentaPromedio = unidadesTotal > 0 ? (ventasTotal / unidadesTotal) : 0;

                // Skip lot if both costs and sales are zero and we filtered by dates/crop
                if ((fechaDesde || fechaHasta || selectedCultivo !== "Todos") && costoTotal === 0 && ventasTotal === 0) {
                    return;
                }

                renderedCount++;

                const tr = document.createElement("tr");
                tr.className = "hover:bg-slate-50/70 dark:hover:bg-slate-800/70 transition duration-150";
                tr.innerHTML = `
                    <td class="px-6 py-4 whitespace-nowrap font-semibold text-slate-900 dark:text-white">${lote}</td>
                    <td class="px-6 py-4 whitespace-nowrap"><span class="inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-semibold bg-emerald-50 text-emerald-800 dark:bg-emerald-950 dark:text-emerald-300">${cultivo}</span></td>
                    <td class="px-6 py-4 whitespace-nowrap text-right text-slate-600 dark:text-slate-300 font-medium">${formatCOP(costoTotal)}</td>
                    <td class="px-6 py-4 whitespace-nowrap text-right text-slate-600 dark:text-slate-300 font-medium">${formatCOP(ventasTotal)}</td>
                    <td class="px-6 py-4 whitespace-nowrap text-right text-slate-600 dark:text-slate-300 font-medium">${unidadesTotal}</td>
                    <td class="px-6 py-4 whitespace-nowrap text-right text-slate-600 dark:text-slate-300 font-medium">${unidadesTotal > 0 ? formatCOP(costoUnitario) : "-"}</td>
                    <td class="px-6 py-4 whitespace-nowrap text-right text-slate-600 dark:text-slate-300 font-medium">${unidadesTotal > 0 ? formatCOP(precioVentaPromedio) : "-"}</td>
                    <td class="px-6 py-4 whitespace-nowrap text-right font-extrabold ${ganancia >= 0 ? 'text-emerald-600 dark:text-emerald-400' : 'text-rose-600 dark:text-rose-400'}">${formatCOP(ganancia)}</td>
                    <td class="px-6 py-4 whitespace-nowrap text-right font-bold ${roi >= 0 ? 'text-emerald-600 dark:text-emerald-400' : 'text-rose-600 dark:text-rose-400'}">${roi.toFixed(1)}%</td>
                `;
                tableBody.appendChild(tr);
            });

            const tablaCountEl = document.getElementById("tablaCount"); if (tablaCountEl) tablaCountEl.textContent = `Mostrando ${renderedCount} ${renderedCount === 1 ? 'lote' : 'lotes'}`;
        }

        function renderCashflowModule() {
            if (!rawData) return;

            const selectedLote = document.getElementById("cashflowLoteSelector")?.value || "Todos";

            const filterByLote = (item) => {
                return isLoteMatch(item.Lote || item.lote, selectedLote);
            };

            const filteredVentas = (rawData.ventas || []).filter(filterByLote);
            const filteredIniciales = (rawData.gastosIniciales || []).filter(filterByLote);
            const filteredDesarrollo = (rawData.gastosDesarrollo || []).filter(filterByLote);

            // 1. Efectivo Ingresado (Caja Real)
            const cajaReal = filteredVentas
                .filter(x => String(x.Estado_Pago || x.estado_pago || 'PAGADO').toUpperCase() === 'PAGADO')
                .reduce((sum, x) => sum + (Number(x.Ingreso_Total_COP || x.ingreso_total_cop) || 0), 0);

            // 2. Efectivo Gastado (Salida Real)
            const salidaReal = filteredIniciales.concat(filteredDesarrollo)
                .filter(x => String(x.Estado_Pago || x.estado_pago || 'PAGADO').toUpperCase() === 'PAGADO')
                .reduce((sum, x) => sum + (Number(x.Costo_Total_COP || x.costo_total_cop) || 0), 0);

            // 3. Saldo Disponible en Caja
            const saldoDisponible = cajaReal - salidaReal;

            // 4. Pending balances
            const cuentasPorCobrar = filteredVentas
                .filter(x => String(x.Estado_Pago || x.estado_pago || '').toUpperCase() === 'PENDIENTE')
                .reduce((sum, x) => sum + (Number(x.Ingreso_Total_COP || x.ingreso_total_cop) || 0), 0);

            const cuentasPorPagar = filteredIniciales.concat(filteredDesarrollo)
                .filter(x => String(x.Estado_Pago || x.estado_pago || '').toUpperCase() === 'PENDIENTE')
                .reduce((sum, x) => sum + (Number(x.Costo_Total_COP || x.costo_total_cop) || 0), 0);

            // 5. Saldo Proyectado
            const saldoProyectado = (cajaReal + cuentasPorCobrar) - cuentasPorPagar;

            // Update UI elements

        function renderCuentasPendientes() {
            if (!rawData) return;

            const tableBody = document.getElementById("cuentasPendientesTableBody");
            if (!tableBody) return;
            tableBody.innerHTML = "";

            let countVencidas = 0;
            let valVencidas = 0;
            let countPorCobrar = 0;
            let valPorCobrar = 0;
            let countPorPagar = 0;
            let valPorPagar = 0;

            const todayStr = new Date().toISOString().split('T')[0];
            let pendingItems = [];

            // 1. Pending Sales (Cuentas por Cobrar)
            (rawData.ventas || []).forEach((v, idx) => {
                const estado = String(v.Estado_Pago || v.estado_pago || '').toUpperCase();
                if (estado === 'PENDIENTE') {
                    const monto = Number(v.Ingreso_Total_COP || v.ingreso_total_cop) || 0;
                    const venc = String(v.Fecha_Vencimiento || v.fecha_vencimiento || '').trim();
                    const isVencida = venc && venc < todayStr;

                    countPorCobrar++;
                    valPorCobrar += monto;

                    if (isVencida) {
                        countVencidas++;
                        valVencidas += monto;
                    }

                    pendingItems.push({
                        tipo: 'venta',
                        tipoText: 'Cobro (Venta)',
                        lote: v.Lote || v.lote || '-',
                        tercero: v.Cliente_Deudor_Proveedor || v.cliente_deudor_proveedor || v.Comprador || v.comprador || 'Cliente Varios',
                        concepto: v.Producto_Vendido || v.producto_vendido || 'Venta de Cosecha',
                        monto: monto,
                        vencimiento: venc || '-',
                        isVencida: isVencida,
                        code: v.Codigo_Seguimiento || v.codigo_seguimiento || (idx + 2)
                    });
                }
            });

            // 2. Pending Initial Expenses (Cuentas por Pagar)
            (rawData.gastosIniciales || []).forEach((ini, idx) => {
                const estado = String(ini.Estado_Pago || ini.estado_pago || '').toUpperCase();
                if (estado === 'PENDIENTE') {
                    const monto = Number(ini.Costo_Total_COP || ini.costo_total_cop) || 0;
                    const venc = String(ini.Fecha_Vencimiento || ini.fecha_vencimiento || '').trim();

                    countPorPagar++;
                    valPorPagar += monto;

                    pendingItems.push({
                        tipo: 'gastosIniciales',
                        tipoText: 'Pago (Gasto Inicial)',
                        lote: ini.Lote || ini.lote || '-',
                        tercero: ini.Cliente_Deudor_Proveedor || ini.cliente_deudor_proveedor || 'Proveedor Varios',
                        concepto: ini.Cultivo_Semilla || ini.cultivo_semilla || 'Establecimiento',
                        monto: monto,
                        vencimiento: venc || '-',
                        isVencida: venc && venc < todayStr,
                        code: ini.Codigo_Seguimiento || ini.codigo_seguimiento || (idx + 2)
                    });
                }
            });

            // 3. Pending Development Expenses (Cuentas por Pagar)
            (rawData.gastosDesarrollo || []).forEach((des, idx) => {
                const estado = String(des.Estado_Pago || des.estado_pago || '').toUpperCase();
                if (estado === 'PENDIENTE') {
                    const monto = Number(des.Costo_Total_COP || des.costo_total_cop) || 0;
                    const venc = String(des.Fecha_Vencimiento || des.fecha_vencimiento || '').trim();

                    countPorPagar++;
                    valPorPagar += monto;

                    pendingItems.push({
                        tipo: 'gastosDesarrollo',
                        tipoText: 'Pago (Gasto Desarrollo)',
                        lote: des.Lote || des.lote || '-',
                        tercero: des.Cliente_Deudor_Proveedor || des.cliente_deudor_proveedor || 'Proveedor/Trabajador',
                        concepto: des.Tipo_Insumo || des.tipo_insumo || 'Mantenimiento',
                        monto: monto,
                        vencimiento: venc || '-',
                        isVencida: venc && venc < todayStr,
                        code: des.Codigo_Seguimiento || des.codigo_seguimiento || (idx + 2)
                    });
                }
            });

            const countCobrarVencidasEl = document.getElementById("countCobrarVencidas");
            if (countCobrarVencidasEl) countCobrarVencidasEl.textContent = countVencidas;
            const valCobrarVencidasEl = document.getElementById("valCobrarVencidas");
            if (valCobrarVencidasEl) valCobrarVencidasEl.textContent = formatCOP(valVencidas);

            const countPorCobrarEl = document.getElementById("countPorCobrar");
            if (countPorCobrarEl) countPorCobrarEl.textContent = countPorCobrar;
            const valPorCobrarEl = document.getElementById("valPorCobrar");

            const valPorPagarEl = document.getElementById("valPorPagar");
            if (valPorPagarEl) valPorPagarEl.textContent = formatCOP(valPorPagar);

            const totalPending = countPorCobrar + countPorPagar;
            const pendingAlertsBadgeEl = document.getElementById("pendingAlertsBadge");
            if (pendingAlertsBadgeEl) pendingAlertsBadgeEl.textContent = `${totalPending} Pendientes`;

            if (pendingItems.length === 0) {

                    <tr>
                        <td colspan="8" class="px-6 py-6 text-center text-xs text-slate-400 font-semibold italic">

        }

        function marcarComoPagado(tipoTabla, filaCodeOrId) {
            if (!confirm("¿Confirmas que deseas marcar esta cuenta pendiente como PAGADA?")) return;

            showLoader(true);
            const fechaHoy = new Date().toISOString().split('T')[0];

            const updateArrayLocally = (arr) => {
                if (!arr) return;
                const match = arr.find(x => 
                    String(x.Codigo_Seguimiento || x.codigo_seguimiento || '').toUpperCase() === String(filaCodeOrId).toUpperCase() ||
                    String(x.Codigo_General || x.codigo_general || '').toUpperCase() === String(filaCodeOrId).toUpperCase()
                );
                if (match) {
                    match.Estado_Pago = 'PAGADO';
                    match.estado_pago = 'PAGADO';
                    match.Fecha_Pago_Realizado = fechaHoy;
                }
            };

            if (tipoTabla === 'venta' || tipoTabla === 'ventas') updateArrayLocally(rawData.ventas);
            if (tipoTabla === 'gastosIniciales') updateArrayLocally(rawData.gastosIniciales);
            if (tipoTabla === 'gastosDesarrollo') updateArrayLocally(rawData.gastosDesarrollo);

            if (typeof google === 'undefined' || !google.script || !google.script.run) {
                setTimeout(() => {
                    showLoader(false);
                    applyFilters();
                    alert("¡Cuenta actualizada como PAGADA exitosamente!");
                }, 600);
            } else {
                google.script.run
                    .withSuccessHandler(function(res) {
                        showLoader(false);
                        applyFilters();
                        alert("¡Estado de pago actualizado exitosamente en Google Sheets!");
                    })
                    .withFailureHandler(function(err) {
                        showLoader(false);
                        alert("Error actualizando en Google Sheets: " + err.toString());
                    })
                    .actualizarEstadoPago(tipoTabla, filaCodeOrId, 'PAGADO', fechaHoy);
            }
        }

        function getLocalActiveLotesList() {
            const list = [];
            if (rawData && rawData.ciclos) {
                (rawData.ciclos || []).forEach(c => {
                    const st = String(c.Estado || c.estado || '').toUpperCase();
                    if (st === 'EN PROCESO' || st === 'ACTIVO') {
                        const lp = String(c.Lote_Principal || c.lote_principal || c.Lote || c.lote || '').trim();
                        const m = lp.match(/^(Lote\s*\d+)/i);
                        const name = m ? m[1] : lp;
                        if (name && !list.includes(name)) list.push(name);
                    }
                });
            }
            if (list.length === 0) {
                for (let i = 1; i <= 16; i++) list.push(`Lote ${i}`);
            }
            return list;
        }

        function startLoteCycle() {
            const lote = document.getElementById("posLote").value;
            const cultivo = document.getElementById("cycleCultivo").value;

            if (!lote || !cultivo) {
                alert("Por favor selecciona un lote y un cultivo.");
                return;
            }

            if (!rawData) rawData = {};
            if (!rawData.ciclos) rawData.ciclos = [];
            const existingActive = (rawData.ciclos || []).find(item => String(item.Lote || item.lote).trim() === lote && String(item.Estado || item.estado).toUpperCase() === 'EN PROCESO');

            if (existingActive) {
                alert(`⚠️ ALERTA DE BLOQUEO: El ${lote} ya cuenta con un ciclo activo en proceso para "${existingActive.Cultivo || existingActive.cultivo}" (Código: ${existingActive.Codigo_General || existingActive.codigo_general}). No se puede iniciar un nuevo ciclo hasta finalizar el actual.`);
                return;
            }

            if (!confirm(`¿Estás seguro de iniciar un ciclo de ${cultivo} en el ${lote}?`)) return;

            showLoader(true, "Iniciando ciclo de cultivo...");

            // Calcular número de ciclo para el lote
            const lotCyclesCount = (rawData.ciclos || []).filter(item => String(item.Lote || item.lote).trim() === lote).length;
            const cycleNum = lotCyclesCount + 1;
            const cycleSuffix = String(cycleNum).padStart(2, '0');

            const cleanLote = lote.replace(/\s+/g, "").toUpperCase();
            const cleanCultivo = cultivo.replace(/\s+/g, "").toUpperCase();
            const year = new Date().getFullYear();

            // Generar un único código de seguimiento para el ciclo del lote (INI-<LOTE>-<CULTIVO>-<YEAR>)
            const singleCycleCode = `INI-${cleanLote}-${cleanCultivo}-${year}`;

            const codigoGeneral = singleCycleCode;
            const codigoInicial = singleCycleCode;
            const codigoDesarrollo = singleCycleCode;
            const codigoVentas = singleCycleCode;
            const fechaInicio = new Date().toISOString().split('T')[0];

            const record = {
                Lote: lote,
                Cultivo: cultivo,
                Codigo_General: codigoGeneral,
                Codigo_Inicial: codigoInicial,
                Codigo_Desarrollo: codigoDesarrollo,
                Codigo_Ventas: codigoVentas,
                Estado: "EN PROCESO",
                Fecha_Inicio: fechaInicio,
                Fecha_Fin: "-"
            };

            if (typeof google === 'undefined' || !google.script || !google.script.run) {
                // Modo Simulado
                setTimeout(() => {
                    if (!rawData.ciclos) rawData.ciclos = [];
                    (rawData.ciclos || []).push(record);
                    updateCyclesTabUI();
                    checkPOSLoteSelection();
                    updateLoteCycleSummaryCard();
                    autoFillActiveCycleCodes();
                    showLoader(false);
                    alert(`¡Ciclo iniciado con éxito (Simulado)! Código General: ${codigoGeneral}`);
                }, 1000);
            } else {
                // Modo Real Google Sheets
                google.script.run
                    .withSuccessHandler(function(response) {
                        showLoader(false);
                        if (response && response.success) {
                            if (!rawData.ciclos) rawData.ciclos = [];
                            (rawData.ciclos || []).push(record);
                            updateCyclesTabUI();
                            checkPOSLoteSelection();
                            updateLoteCycleSummaryCard();
                            autoFillActiveCycleCodes();
                            alert(`¡Ciclo iniciado con éxito en Google Sheets! Código General: ${codigoGeneral}`);
                        } else {
                            alert(response ? response.error : "Error de respuesta del servidor.");
                        }
                    })
                    .withFailureHandler(function(err) {
                        showLoader(false);
                        alert("Error al iniciar ciclo: " + err.toString());
                    })
                    .agregarCiclo(lote, cultivo, codigoGeneral, codigoInicial, codigoDesarrollo, codigoVentas, "EN PROCESO", fechaInicio, "-");
            }
        }

        function endLoteCycleDirect(codigoGeneral, fechaFin, callback) {
            const fFin = fechaFin || new Date().toISOString().split('T')[0];
            const codeUpper = (codigoGeneral || "").toString().trim().toUpperCase();

            const updateLocalCycleState = () => {
                const cycles = rawData.ciclos || [];
                let cycle = cycles.find(item => {
                    const cg = (item.Codigo_General || item.codigo_general || "").toString().trim().toUpperCase();
                    const ci = (item.Codigo_Inicial || item.codigo_inicial || "").toString().trim().toUpperCase();
                    const cd = (item.Codigo_Desarrollo || item.codigo_desarrollo || "").toString().trim().toUpperCase();
                    const cv = (item.Codigo_Ventas || item.codigo_ventas || "").toString().trim().toUpperCase();
                    return (cg === codeUpper || ci === codeUpper || cd === codeUpper || cv === codeUpper) && String(item.Estado || item.estado).toUpperCase() === "EN PROCESO";
                });

                if (!cycle && codeUpper) {
                    cycle = cycles.find(item => {
                        const cg = (item.Codigo_General || item.codigo_general || "").toString().trim().toUpperCase();
                        const ci = (item.Codigo_Inicial || item.codigo_inicial || "").toString().trim().toUpperCase();
                        const cd = (item.Codigo_Desarrollo || item.codigo_desarrollo || "").toString().trim().toUpperCase();
                        const cv = (item.Codigo_Ventas || item.codigo_ventas || "").toString().trim().toUpperCase();
                        return cg === codeUpper || ci === codeUpper || cd === codeUpper || cv === codeUpper;
                    });
                }

                if (cycle) {
                    cycle.Estado = "FINALIZADO";
                    cycle.Fecha_Fin = fFin;
                }

                updateCyclesTabUI();
                checkPOSLoteSelection();
                updateLoteCycleSummaryCard();
                autoFillActiveCycleCodes();
                if (typeof renderInteractiveMap === 'function') renderInteractiveMap();
                if (typeof applyFilters === 'function') applyFilters();
            };

            if (typeof google === 'undefined' || !google.script || !google.script.run) {
                updateLocalCycleState();
                if (callback) callback();
            } else {
                google.script.run
                    .withSuccessHandler(function(response) {
                        if (response && response.success) {
                            updateLocalCycleState();
                        } else {
                            alert(response ? response.error : "No se pudo finalizar el ciclo.");
                        }
                        if (callback) callback();
                    })
                    .withFailureHandler(function(err) {
                        alert("Error al cerrar ciclo de seguimiento: " + err.toString());
                        if (callback) callback();
                    })
                    .finalizarCiclo(codigoGeneral, fFin);
            }
        }

        function endLoteCycle(codigoGeneral) {
            if (!confirm(`¿Estás seguro de finalizar el ciclo con código ${codigoGeneral}? Esto liberará el lote.`)) return;

            showLoader(true, "Finalizando ciclo de cultivo...");
            const fechaFin = new Date().toISOString().split('T')[0];

            endLoteCycleDirect(codigoGeneral, fechaFin, () => {
                showLoader(false);
                setTimeout(() => {
                    alert("¡Ciclo finalizado y lote liberado con éxito!");
                }, 50);
            });
        }

                summaryCard.classList.add("hidden");

        function autoFillActiveCycleCodes() {
            const lote = document.getElementById("posLote").value;
            const codeInput = document.getElementById("posTrackingCode");
            if (!codeInput) return;

            if (!lote) {
                codeInput.value = "";
                codeInput.readOnly = false;
                return;
            }

            if (!rawData) rawData = {};
            if (!rawData.ciclos) rawData.ciclos = [];
            const activeCycle = (rawData.ciclos || []).find(item => String(item.Lote || item.lote).trim() === lote && String(item.Estado || item.estado).toUpperCase() === 'EN PROCESO');

            if (activeCycle) {
                let codeToFill = activeCycle.Codigo_General || activeCycle.codigo_general || activeCycle.Codigo_Inicial || activeCycle.codigo_inicial;

                if (codeToFill) {
                    codeInput.value = codeToFill;
                    codeInput.readOnly = true;
                    showPOSAlert(`Código de seguimiento vinculado al ciclo activo: ${codeToFill}`, "success");
                } else {
                    codeInput.readOnly = false;
                }

                // Auto-select active crop and sale product
                const activeCrop = activeCycle.Cultivo || activeCycle.cultivo;
                if (activeCrop) {
                    const cultivoSelect = document.getElementById("posCultivo");
                    if (cultivoSelect && activePOSTab === 'gastos_iniciales') {
                        if ([...cultivoSelect.options].some(o => o.value === activeCrop)) {
                            cultivoSelect.value = activeCrop;
                        }
                    }

                    const productoSelect = document.getElementById("posProducto");
                    if (productoSelect && activePOSTab === 'ventas') {
                        const productMap = {
                            "Café Castillo": "Café Pergamino Seco",
                            "Aguacate Hass": "Aguacate Hass Extra",
                            "Cacao Fino de Aroma": "Cacao en Grano",
                            "Limón Tahití": "Limón Tahití Selección",
                            "Plátano Hartón": "Plátano Hartón de Primera"
                        };
                        const expectedProduct = productMap[activeCrop];
                        if (expectedProduct && [...productoSelect.options].some(o => o.value === expectedProduct)) {
                            productoSelect.value = expectedProduct;
                        }
                    }
                }
            } else {
                codeInput.readOnly = false;
            }
        }

        function autoGenerateTrackingCode() {
            const lote = document.getElementById("posLote").value;
            if (!lote) {
                showPOSAlert("Selecciona primero un lote para generar el código.", "error");
                return;

        }

        function parseDecimal(val) {
            if (val === null || val === undefined || val === '') return 0;
            const strVal = String(val).replace(',', '.');
            const parsed = parseFloat(strVal);
            return isNaN(parsed) ? 0 : parsed;
        }

        function calcPOSTotals(section, triggerSource) {
            let qtyEl, unitEl, totalEl;

            if (section === 'inicial') {
                qtyEl = document.getElementById("posCantidadInicial");
                unitEl = document.getElementById("posCostoUnitarioInicial");
                totalEl = document.getElementById("posCostoInicial");
            } else if (section === 'desarrollo') {
                qtyEl = document.getElementById("posCantidadDesarrollo");
                unitEl = document.getElementById("posCostoUnitarioDesarrollo");
                totalEl = document.getElementById("posCostoDesarrollo");
            } else if (section === 'venta') {
                qtyEl = document.getElementById("posUnidadesVenta");
                unitEl = document.getElementById("posPrecioUnitarioVenta");
                totalEl = document.getElementById("posIngresoVenta");
            }

            if (!qtyEl || !unitEl || !totalEl) return;

            const qty = parseDecimal(qtyEl.value) || 1;
            const unit = parseDecimal(unitEl.value);
            const total = parseDecimal(totalEl.value);

            if (triggerSource === 'unitario') {
                totalEl.value = Math.round((qty * unit) * 100) / 100;
            } else if (triggerSource === 'total') {
                if (qty > 0) {
                    unitEl.value = Math.round((total / qty) * 100) / 100;
                }
            }
        }

        function togglePOSDueDate(section) {
            let radioName = "";
            let containerId = "";
            let inputId = "";

            if (section === 'inicial') {
                radioName = "posEstadoPagoInicial";
                containerId = "posContainerFechaVencInicial";
                inputId = "posFechaVencimientoInicial";
            } else if (section === 'desarrollo') {
                radioName = "posEstadoPagoDesarrollo";
                containerId = "posContainerFechaVencDesarrollo";
                inputId = "posFechaVencimientoDesarrollo";
            } else if (section === 'venta') {
                radioName = "posEstadoPagoVenta";
                containerId = "posContainerFechaVencVenta";
                inputId = "posFechaVencimientoVenta";
            }

            const radios = document.getElementsByName(radioName);
            let selectedVal = "PAGADO";
            for (let i = 0; i < radios.length; i++) {
                if (radios[i].checked) {
                    selectedVal = radios[i].value;
                    break;
                }
            }

            const container = document.getElementById(containerId);
            const input = document.getElementById(inputId);

            if (container && input) {
                if (selectedVal === 'PENDIENTE') {
                    container.classList.remove("opacity-50", "pointer-events-none");
                    input.disabled = false;
                    if (!input.value) {
                        const d = new Date();
                        d.setDate(d.getDate() + 15);
                        input.value = d.toISOString().split('T')[0];
                    }
                } else {
                    container.classList.add("opacity-50", "pointer-events-none");
                    input.disabled = true;
                }
            }
        }

            const ventasCobradas = (rawData.ventas || [])
                .filter(x => String(x.Estado_Pago || x.estado_pago || 'PAGADO').toUpperCase() === 'PAGADO')
                .reduce((sum, x) => sum + (Number(x.Ingreso_Total_COP || x.ingreso_total_cop) || 0), 0);

            const gastosAgricolasPagados = (rawData.gastosIniciales || []).concat(rawData.gastosDesarrollo || [])

        function renderMetasCards() {
            const container = document.getElementById("metasCardsContainer");
            if (!container) return;

            container.innerHTML = "";
            const metas = rawData.metasRecoleccion || [];

            if (metas.length === 0) {
                container.innerHTML = `
                    <div class="col-span-full text-center py-8 text-slate-400 font-semibold text-xs">
                        <svg class="h-10 w-10 mx-auto text-slate-300 dark:text-slate-600 mb-2" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                            <path stroke-linecap="round" stroke-linejoin="round" stroke-width="1.5" d="M13 10V3L4 14h7v7l9-11h-7z"></path>
                        </svg>
                        No hay metas de recolección registradas. Crea una meta arriba para iniciar la medición de ritmo y plazo.
                    </div>
                `;
                return;
            }

            const today = new Date();
            today.setHours(0,0,0,0);

            metas.forEach(m => {
                const idMeta = m.ID_Meta || m.id_meta;
                const nombre = m.Nombre_Meta || m.nombre_meta || 'Meta de Recolección';
                const montoObj = Number(m.Monto_Objetivo_COP || m.monto_objetivo_cop || m.Monto_Objetivo || m.monto_objetivo) || 0;
                const fechaLimiteStr = m.Fecha_Limite || m.fecha_limite || "";
                const categoria = m.Categoria || m.categoria || "Gasto General";
                const estado = String(m.Estado || m.estado || "EN PROCESO").toUpperCase();

                // Sum all contributions for this specific goal
                const recaudado = (rawData.aportesMetas || [])
                    .filter(a => (a.ID_Meta || a.id_meta) === idMeta)
                    .reduce((sum, x) => sum + (Number(x.Monto_Aporte_COP || x.monto_aporte_cop || x.Monto || x.monto) || 0), 0);

                const faltante = Math.max(0, montoObj - recaudado);
                const porcentaje = montoObj > 0 ? Math.min(100, Math.round((recaudado / montoObj) * 100)) : 0;

                // Time remaining calculations
                let diasRestantes = 0;
                let semanasRestantes = 0;
                let ritmoDiario = 0;
                let ritmoSemanal = 0;
                let fechaLimiteText = fechaLimiteStr || "Sin definir";

                if (fechaLimiteStr) {
                    const parts = fechaLimiteStr.split('-');
                    if (parts.length === 3) {
                        const limiteDate = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]));
                        limiteDate.setHours(0,0,0,0);
                        const diffTime = limiteDate.getTime() - today.getTime();
                        diasRestantes = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
                    }
                }

                if (diasRestantes > 0 && faltante > 0) {
                    ritmoDiario = faltante / diasRestantes;
                    semanasRestantes = Math.max(1, diasRestantes / 7);
                    ritmoSemanal = faltante / semanasRestantes;
                }

        function exportPersonalExcel() {
            let csvContent = "\uFEFF";
            csvContent += "BALANCE DE CAJA GENERAL Y GASTOS DEL HOGAR - FINCA LOS SAUCES\n";
            csvContent += `Fecha del Reporte: ${new Date().toLocaleDateString('es-CO')}\n\n`;

            csvContent += "Métrica;Valor\n";
            csvContent += `Caja Real Agrícola;${document.getElementById("kpiCajaRealAgricola")?.textContent || "$0"}\n`;
            csvContent += `Gastos Personales / Hogar;${document.getElementById("kpiGastosPersonales")?.textContent || "$0"}\n`;
            csvContent += `Saldo Líquido Disponible;${document.getElementById("kpiSaldoLiquido")?.textContent || "$0"}\n\n`;

            csvContent += "Fecha;Categoría;Descripción / Detalle;Monto (COP);Método Pago;Beneficiario / Tercero\n";
            (rawData.gastosPersonales || []).forEach(p => {
                const row = [
                    p.Fecha || p.fecha || '-',
                    p.Categoria || p.categoria || 'Varios',
                    (p.Descripcion_Detalle || p.descripcion_detalle || p.Detalle || '-').replace(/;/g, ','),
                    formatCOP(Number(p.Monto_COP || p.monto_cop) || 0).replace(/;/g, ''),
                    p.Metodo_Pago || p.metodo_pago || 'Efectivo',
                    (p.Beneficiario_Tercero || p.beneficiario_tercero || '-').replace(/;/g, ',')
                ];
                csvContent += row.join(';') + "\n";
            });

            const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement("a");

        function executeLocalMemoryRemoval(tipoTabla, codeOrId, recordInfo) {
            if (!rawData) return;
            const targetCode = String(codeOrId || '').trim().toUpperCase();

            const isTargetMatch = (item) => {
                if (!item) return false;
                const c = String(item.Codigo_Seguimiento || item.codigo_seguimiento || '').trim().toUpperCase();
                const g = String(item.Codigo_General || item.codigo_general || '').trim().toUpperCase();

                if (targetCode && targetCode !== "-" && targetCode !== "SIN CÓDIGO") {
                    if (c === targetCode || g === targetCode) return true;
                }

                if (recordInfo) {
                    const itemLote = String(item.Lote || item.lote || '').trim().toUpperCase();
                    const itemFecha = String(item.Fecha || item.fecha || '').trim();
                    const itemConcepto = String(item.Cultivo_Semilla || item.Tipo_Insumo || item.Producto_Vendido || item.cultivo_semilla || item.tipo_insumo || item.producto_vendido || '').trim().toUpperCase();

                    if (recordInfo.lote && itemLote === recordInfo.lote.toUpperCase() &&
                        recordInfo.fecha && itemFecha === recordInfo.fecha &&
                        recordInfo.concepto && itemConcepto === recordInfo.concepto.toUpperCase()) {
                        return true;
                    }
                }
                return false;
            };

            let foundIdx = -1;

            let loteFound = recordInfo ? recordInfo.lote : "";

            if (tipoTabla === 'gastosIniciales' || tipoTabla === 'inicial') {
                foundIdx = (rawData.gastosIniciales || []).findIndex(isTargetMatch);
                if (foundIdx !== -1) {
                    if (!loteFound) loteFound = rawData.gastosIniciales[foundIdx].Lote || rawData.gastosIniciales[foundIdx].lote;
                    (rawData.gastosIniciales || []).splice(foundIdx, 1);
                }
            } else if (tipoTabla === 'gastosDesarrollo' || tipoTabla === 'desarrollo') {
                foundIdx = (rawData.gastosDesarrollo || []).findIndex(isTargetMatch);
                if (foundIdx !== -1) {
                    if (!loteFound) loteFound = rawData.gastosDesarrollo[foundIdx].Lote || rawData.gastosDesarrollo[foundIdx].lote;
                    (rawData.gastosDesarrollo || []).splice(foundIdx, 1);
                }
            } else if (tipoTabla === 'ventas' || tipoTabla === 'venta') {
                foundIdx = (rawData.ventas || []).findIndex(isTargetMatch);
                if (foundIdx !== -1) {
                    if (!loteFound) loteFound = rawData.ventas[foundIdx].Lote || rawData.ventas[foundIdx].lote;
                    (rawData.ventas || []).splice(foundIdx, 1);
                }
            } else if (tipoTabla === 'ingresosExternos' || tipoTabla === 'ingresoExterno') {
                foundIdx = (rawData.ingresosExternos || []).findIndex(x => (x.ID_Ingreso_Externo || x.id_ingreso_externo || x.id) === codeOrId);
                if (foundIdx !== -1) (rawData.ingresosExternos || []).splice(foundIdx, 1);
            } else if (tipoTabla === 'metasRecoleccion' || tipoTabla === 'metaRecoleccion') {
                foundIdx = (rawData.metasRecoleccion || []).findIndex(x => (x.ID_Meta || x.id_meta || x.id) === codeOrId);
                if (foundIdx !== -1) (rawData.metasRecoleccion || []).splice(foundIdx, 1);
            } else if (tipoTabla === 'aportesMetas' || tipoTabla === 'aporteMeta') {
                foundIdx = (rawData.aportesMetas || []).findIndex(x => (x.ID_Aporte || x.id_aporte || x.id) === codeOrId);
                if (foundIdx !== -1) (rawData.aportesMetas || []).splice(foundIdx, 1);
            }

            // Sync memory cycles: if no remaining transaction exists for targetCode or lot, remove cycle entry
            if (rawData.ciclos && (targetCode || loteFound)) {
                const targetLoteClean = (loteFound || "").trim().toUpperCase();
                const codeUpper = (targetCode || "").trim().toUpperCase();

                const hasCodeTx = (arr) => (arr || []).some(x => {
                    const c = String(x.Codigo_Seguimiento || x.codigo_seguimiento || x.Codigo_General || x.codigo_general || '').trim().toUpperCase();
                    return codeUpper && c === codeUpper;
                });

                const hasLoteTx = (arr) => (arr || []).some(x => {
                    const l = String(x.Lote || x.lote || x.Lote_Principal || '').trim().toUpperCase();
                    return targetLoteClean && (l === targetLoteClean || l.indexOf(targetLoteClean) === 0);
                });

                const codeHasRem = hasCodeTx(rawData.gastosIniciales) || hasCodeTx(rawData.gastosDesarrollo) || hasCodeTx(rawData.ventas);
                const loteHasRem = hasLoteTx(rawData.gastosIniciales) || hasLoteTx(rawData.gastosDesarrollo) || hasLoteTx(rawData.ventas);

                rawData.ciclos = (rawData.ciclos || []).filter(c => {
                    const cCode = String(c.Codigo_General || c.codigo_general || c.Codigo_Inicial || c.codigo_inicial || '').trim().toUpperCase();
                    const cLote = String(c.Lote || c.lote || c.Lote_Principal || '').trim().toUpperCase();

                    const isMatchCode = codeUpper && cCode === codeUpper;
                    const isMatchLote = targetLoteClean && (cLote === targetLoteClean || cLote.indexOf(targetLoteClean) === 0);

                    if ((isMatchCode && !codeHasRem) || (isMatchLote && !loteHasRem)) {
                        return false; // delete cycle
                    }
                    return true;
                });
            }
        }

        function filtrarTransaccionesPorLote(idLote) {
            if (!rawData) return [];
            const targetLote = String(idLote || "Todos").trim();
            const codeToGeneralMap = {};

            if (rawData.ciclos) {
                (rawData.ciclos || []).forEach(c => {
                    const gral = String(c.Codigo_General || c.codigo_general || "").toUpperCase().trim();
                    const ini = String(c.Codigo_Inicial || c.codigo_inicial || "").toUpperCase().trim();
                    const des = String(c.Codigo_Desarrollo || c.codigo_desarrollo || "").toUpperCase().trim();
                    const ven = String(c.Codigo_Ventas || c.codigo_ventas || "").toUpperCase().trim();

                    if (gral) {
                        if (ini) codeToGeneralMap[ini] = gral;
                        if (des) codeToGeneralMap[des] = gral;
                        if (ven) codeToGeneralMap[ven] = gral;
                    }
                });
            }

            let allTx = [];

            const isMatch = (itemLoteStr) => {
                return isLoteMatch(itemLoteStr, targetLote);
            };

            // 1. Gastos Iniciales
            (rawData.gastosIniciales || []).forEach(item => {
                const itemLote = item.Lote || item.lote || "Sin Registrar";
                if (isMatch(itemLote)) {
                    const code = (item.Codigo_Seguimiento || item.codigo_seguimiento || "-").trim().toUpperCase();
                    allTx.push({
                        lote: itemLote,
                        tipo: "Inicial",
                        tipoTable: "gastosIniciales",
                        tipoLabel: "Gasto Inicial",
                        concepto: item.Cultivo_Semilla || item.cultivo_semilla || "Por Definir",
                        monto: Number(item.Costo_Total_COP || item.costo_total_cop) || 0,
                        fecha: item.Fecha || item.fecha || "-",
                        codigo: code,
                        codigoGeneral: codeToGeneralMap[code] || "",
                        detalles: item.Notas || item.notas || "-"
                    });
                }
            });

            // 2. Gastos Desarrollo
            (rawData.gastosDesarrollo || []).forEach(item => {
                const itemLote = item.Lote || item.lote || "Sin Registrar";
                if (isMatch(itemLote)) {
                    const code = (item.Codigo_Seguimiento || item.codigo_seguimiento || "-").trim().toUpperCase();
                    allTx.push({
                        lote: itemLote,
                        tipo: "Desarrollo",
                        tipoTable: "gastosDesarrollo",
                        tipoLabel: "Gasto Desarrollo",
                        concepto: item.Tipo_Insumo || item.tipo_insumo || "Otros",
                        monto: Number(item.Costo_Total_COP || item.costo_total_cop) || 0,
                        fecha: item.Fecha || item.fecha || "-",
                        codigo: code,
                        codigoGeneral: codeToGeneralMap[code] || "",
                        detalles: item.Detalle || item.detalle || "-"
                    });
                }
            });

            // 3. Ventas
            (rawData.ventas || []).forEach(item => {
                const itemLote = item.Lote || item.lote || "Sin Registrar";
                if (isMatch(itemLote)) {
                    const code = (item.Codigo_Seguimiento || item.codigo_seguimiento || "-").trim().toUpperCase();
                    allTx.push({
                        lote: itemLote,
                        tipo: "Venta",
                        tipoTable: "ventas",
                        tipoLabel: "Venta",
                        concepto: item.Producto_Vendido || item.producto_vendido || "Otros",
                        monto: Number(item.Ingreso_Total_COP || item.ingreso_total_cop) || 0,
                        fecha: item.Fecha || item.fecha || "-",
                        codigo: code,
                        codigoGeneral: codeToGeneralMap[code] || "",
                        detalles: item.Comprador || item.comprador || "-"
                    });
                }
            });

            return allTx;
        }

        function renderTransaccionesTable(selectedLote) {
            const tableBody = document.getElementById("transaccionesTableBody");
            const countEl = document.getElementById("transaccionesCount");
            if (!tableBody || !rawData) return;
            tableBody.innerHTML = "";

            const targetLote = selectedLote || document.getElementById("filterLote")?.value || "Todos";
            const searchCode = (document.getElementById("searchTrackingCode")?.value || "").trim().toUpperCase();
            const filterType = document.getElementById("filterMovimiento")?.value || "Todos";
            const selectedCiclo = document.getElementById("filterCiclo") ? document.getElementById("filterCiclo").value : "Todos";

            let filtered = filtrarTransaccionesPorLote(targetLote);

            // Filter by Cycle Dropdown if specific cycle chosen
            if (selectedCiclo && selectedCiclo !== "Todos") {
                filtered = filtered.filter(tx => tx.codigo === selectedCiclo.toUpperCase() || tx.codigoGeneral === selectedCiclo.toUpperCase());
            }

            // Filter by Movement Type
            if (filterType !== "Todos") {
                filtered = filtered.filter(tx => tx.tipo === filterType);
            }

            // Filter by Tracking Code
            if (searchCode) {
                filtered = filtered.filter(tx => {
                    const matchesStageCode = tx.codigo.includes(searchCode);
                    const matchesGeneralCode = tx.codigoGeneral && tx.codigoGeneral.includes(searchCode);
                    return matchesStageCode || matchesGeneralCode;
                });

        function renderEstadoResultados() {
            if (!rawData) return;

            const selectedLote = document.getElementById("filterLote") ? document.getElementById("filterLote").value : "Todos";
            const selectedCiclo = document.getElementById("filterCiclo") ? document.getElementById("filterCiclo").value : "Todos";

            const isMatch = (item) => {
                if (!isLoteMatch(item.Lote || item.lote, selectedLote)) return false;
                if (selectedCiclo && selectedCiclo !== "Todos") {
                    const code = String(item.Codigo_Seguimiento || item.codigo_seguimiento || item.Codigo_General || item.codigo_general || "").trim().toUpperCase();
                    if (code !== selectedCiclo.trim().toUpperCase()) return false;
                }
                return true;
            };

            // 1. Calculate P&L Totals according to selected lote and ciclo
            const totalIniciales = (rawData.gastosIniciales || []).filter(isMatch).reduce((sum, item) => sum + (Number(item.Costo_Total_COP || item.costo_total_cop) || 0), 0);
            const totalDesarrollo = (rawData.gastosDesarrollo || []).filter(isMatch).reduce((sum, item) => sum + (Number(item.Costo_Total_COP || item.costo_total_cop) || 0), 0);
            const totalVentas = (rawData.ventas || []).filter(isMatch).reduce((sum, item) => sum + (Number(item.Ingreso_Total_COP || item.ingreso_total_cop) || 0), 0);
            const utilidadNeta = totalVentas - (totalIniciales + totalDesarrollo);

            document.getElementById("plVentas").textContent = formatCOP(totalVentas);
            document.getElementById("plCostosIniciales").textContent = formatCOP(totalIniciales);
            document.getElementById("plCostosDesarrollo").textContent = formatCOP(totalDesarrollo);
            
            const plUtilidadEl = document.getElementById("plUtilidad");
            plUtilidadEl.textContent = formatCOP(utilidadNeta);
            if (utilidadNeta >= 0) {
                plUtilidadEl.className = "text-xl sm:text-2xl font-extrabold text-emerald-600 mt-1";
            } else {
                plUtilidadEl.className = "text-xl sm:text-2xl font-extrabold text-rose-600 mt-1";
            }

            // 2. Populate Dropdown items uniquely from all categories
            const uniqueItems = new Set();
            
            if (rawData.gastosIniciales) {
                (rawData.gastosIniciales || []).forEach(item => {
                    const c = String(item.Cultivo_Semilla || item.cultivo_semilla || "").trim();
                    if (c) uniqueItems.add(c);
                });
            }
            if (rawData.gastosDesarrollo) {
                (rawData.gastosDesarrollo || []).forEach(item => {
                    const ins = String(item.Tipo_Insumo || item.tipo_insumo || "").trim();
                    if (ins) uniqueItems.add(ins);
                });
            }
            if (rawData.ventas) {
                (rawData.ventas || []).forEach(item => {
                    const p = String(item.Producto_Vendido || item.producto_vendido || "").trim();
                    if (p) uniqueItems.add(p);
                });
            }

            const itemSelector = document.getElementById("variationItemSelector");
            if (itemSelector) {
                const prevVal = itemSelector.value;
                itemSelector.innerHTML = "";
                
                const sortedItems = Array.from(uniqueItems).sort();
                sortedItems.forEach(item => {
                    const opt = document.createElement("option");
                    opt.value = item;
                    opt.textContent = item;
                    itemSelector.appendChild(opt);
                });

                if (prevVal && sortedItems.includes(prevVal)) {
                    itemSelector.value = prevVal;
                } else if (sortedItems.length > 0) {
                    itemSelector.value = sortedItems[0];
                }
            }

            renderPriceVariationDetail();
        }

        function renderPriceVariationDetail() {
            const tableBody = document.getElementById("varHistoryTableBody");
            const itemSelector = document.getElementById("variationItemSelector");
            if (!tableBody || !itemSelector || !rawData) return;

            tableBody.innerHTML = "";
            const selectedItem = itemSelector.value;
            if (!selectedItem) {
                document.getElementById("varMin").textContent = "$0";
                document.getElementById("varMax").textContent = "$0";
                document.getElementById("varAverage").textContent = "$0";
                document.getElementById("varPct").textContent = "0%";
                document.getElementById("varTrendBadge").className = "inline-flex items-center px-2 py-0.5 rounded-full text-[10px] font-bold mt-2 bg-slate-100 text-slate-800";
                document.getElementById("varTrendBadge").textContent = "Sin Datos";
                return;
            }

            let history = [];

            // Harvest item prices chronologically
            if (rawData.gastosIniciales) {
                (rawData.gastosIniciales || []).forEach(item => {
                    if (String(item.Cultivo_Semilla || item.cultivo_semilla || "").trim() === selectedItem) {
                        history.push({
                            fecha: item.Fecha || item.fecha || "-",
                            lote: item.Lote || item.lote || "-",
                            valor: Number(item.Costo_Total_COP || item.costo_total_cop) || 0,
                            codigo: (item.Codigo_Seguimiento || item.codigo_seguimiento || "-").trim().toUpperCase()
                        });
                    }
                });
            }

            if (rawData.gastosDesarrollo) {
                (rawData.gastosDesarrollo || []).forEach(item => {
                    if (String(item.Tipo_Insumo || item.tipo_insumo || "").trim() === selectedItem) {
                        history.push({
                            fecha: item.Fecha || item.fecha || "-",
                            lote: item.Lote || item.lote || "-",
                            valor: Number(item.Costo_Total_COP || item.costo_total_cop) || 0,
                            codigo: (item.Codigo_Seguimiento || item.codigo_seguimiento || "-").trim().toUpperCase()
                        });
                    }
                });
            }

            if (rawData.ventas) {
                (rawData.ventas || []).forEach(item => {
                    if (String(item.Producto_Vendido || item.producto_vendido || "").trim() === selectedItem) {
                        history.push({
                            fecha: item.Fecha || item.fecha || "-",
                            lote: item.Lote || item.lote || "-",
                            valor: Number(item.Ingreso_Total_COP || item.ingreso_total_cop) || 0,
                            codigo: (item.Codigo_Seguimiento || item.codigo_seguimiento || "-").trim().toUpperCase()
                        });
                    }
                });
            }

            // Sort chronologically Ascending (from past to present)
            history.sort((a, b) => {
                if (a.fecha === "-" || !a.fecha) return 1;
                if (b.fecha === "-" || !b.fecha) return -1;
                return a.fecha.localeCompare(b.fecha);
            });

            // Calculate metrics
            let minVal = Infinity;
            let maxVal = -Infinity;
            let sumVal = 0;
            
            history.forEach(tx => {
                if (tx.valor < minVal) minVal = tx.valor;
                if (tx.valor > maxVal) maxVal = tx.valor;
                sumVal += tx.valor;
            });

            const count = history.length;
            if (count === 0) {
                minVal = 0;
                maxVal = 0;
            }
            const avgVal = count > 0 ? sumVal / count : 0;
            
            // Calculate variation: ((Max - Min) / Min) * 100
            let pctVar = 0;
            if (count > 1 && minVal > 0 && minVal !== Infinity) {
                pctVar = ((maxVal - minVal) / minVal) * 100;
            }

            // Render history table rows
            history.forEach(tx => {
                const tr = document.createElement("tr");
                tr.className = "hover:bg-slate-50/70 transition duration-150";

                const displayCode = tx.codigo === "-" || !tx.codigo ? 
                    `<span class="text-slate-400 italic">Sin Código</span>` : 
                    `<span class="px-2 py-0.5 rounded bg-slate-100 text-slate-800 font-mono text-xs border border-slate-200 font-bold">${tx.codigo}</span>`;

                tr.innerHTML = `
                    <td class="px-4 py-3 whitespace-nowrap text-xs text-slate-500 font-medium">${tx.fecha}</td>
                    <td class="px-4 py-3 whitespace-nowrap text-xs text-slate-900 font-bold">${tx.lote}</td>
                    <td class="px-4 py-3 whitespace-nowrap text-xs text-right text-slate-800 font-extrabold">${formatCOP(tx.valor)}</td>
                    <td class="px-4 py-3 whitespace-nowrap">${displayCode}</td>
                `;
                tableBody.appendChild(tr);
            });

            // Update Metrics Cards
            document.getElementById("varMin").textContent = formatCOP(minVal === Infinity ? 0 : minVal);
            document.getElementById("varMax").textContent = formatCOP(maxVal === -Infinity ? 0 : maxVal);
            document.getElementById("varAverage").textContent = formatCOP(avgVal);
            document.getElementById("varPct").textContent = `${Math.round(pctVar)}%`;

            const badge = document.getElementById("varTrendBadge");
            if (count <= 1) {
                badge.className = "inline-flex items-center px-2.5 py-0.5 rounded-full text-[10px] font-bold mt-2 bg-slate-100 text-slate-800";
                badge.textContent = "Estable (1 registro)";
            } else if (pctVar > 10) {
                badge.className = "inline-flex items-center px-2.5 py-0.5 rounded-full text-[10px] font-bold mt-2 bg-rose-50 text-rose-800 border border-rose-200";
                badge.textContent = "Volatilidad Alta";
            } else {
                badge.className = "inline-flex items-center px-2.5 py-0.5 rounded-full text-[10px] font-bold mt-2 bg-emerald-50 text-emerald-800 border border-emerald-200";
                badge.textContent = "Volatilidad Baja";
            }
        }

                    .filter(item => String(item.Lote || item.lote).trim() === lote);

                const precioVentaPromedio = unidadesTotal > 0 ? (ventasTotal / unidadesTotal) : 0;

                if (costoTotal > 0 || ventasTotal > 0) {
                    bodyRows.push({
                        lote: lote,
                        cultivo: cultivo,
                        gastos: formatCOP(costoTotal),
                        ingresos: formatCOP(ventasTotal),
                        unidades: String(unidadesTotal),
                        costo_u: unidadesTotal > 0 ? formatCOP(costoUnitario) : "-",
                        venta_p: unidadesTotal > 0 ? formatCOP(precioVentaPromedio) : "-",
                        utilidad: formatCOP(ganancia),
                        roi: `${roi.toFixed(1)}%`
                    });
                }
            });

            doc.autoTable({
                columns: columns,
                body: bodyRows,
                startY: 220,
                margin: { left: 40, right: 40 },
                styles: {
                    font: 'helvetica',
                    fontSize: 8.5,
                    cellPadding: 5,
                    lineColor: [226, 232, 240],
                    lineWidth: 0.5
                },
                headStyles: {
                    fillColor: [5, 150, 105], // Emerald 600
                    textColor: [255, 255, 255],
                    fontStyle: 'bold'
                },
                alternateRowStyles: {

                },
                columnStyles: {
                    gastos: { halign: 'right' },
                    ingresos: { halign: 'right' },

        function exportDashboardToExcel() {
            const triggerDownload = (csvData, filename) => {
                const blob = new Blob([csvData], { type: 'text/csv;charset=utf-8;' });
                const link = document.createElement("a");
                const url = URL.createObjectURL(blob);
                link.setAttribute("href", url);
                link.setAttribute("download", filename || `Consolidado_General_Finca_Los_Sauces_${new Date().toISOString().split('T')[0]}.csv`);
                link.style.visibility = 'hidden';
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
            };

            showLoader(true, "Generando consolidado para Excel...");

            if (typeof google !== 'undefined' && google.script && google.script.run) {
                google.script.run
                    .withSuccessHandler(function(res) {
                        showLoader(false);
                        if (res && res.success && res.csvContent) {
                            triggerDownload(res.csvContent, res.filename);
                        } else {
                            generateLocalExcelDownload(triggerDownload);
                        }
                    })
                    .withFailureHandler(function(err) {
                        showLoader(false);
                        generateLocalExcelDownload(triggerDownload);
                    })
                    .exportToExcelData();
            } else {
                showLoader(false);
                generateLocalExcelDownload(triggerDownload);
            }
        }

        function calculateStageInfo(fechaInicio, cultivo) {
            const duracion = 120; // Ciclo estándar de 120 días (4 meses)

            if (!fechaInicio || fechaInicio === "-" || fechaInicio === "" || !cultivo || cultivo === "Libre / Barbecho" || cultivo === "Barbecho") {
                return {
                    diasTranscurridos: 0,
                    duracionEstimadaDias: duracion,
                    porcentajeAvance: 0,
                    etapaNombre: "Libre / Barbecho",
                    etapaNumero: 0,
                    colorHex: "#64748b",
                    colorStatus: "gris",
                    estadoTexto: "LIBRE"
                };
            }

            const hoy = new Date();
            const inicio = new Date(fechaInicio);
            const diffTime = Math.max(0, hoy.getTime() - inicio.getTime());
            const diasTranscurridos = Math.floor(diffTime / (1000 * 60 * 60 * 24));
            const porcentajeAvance = Math.min(100, Math.round((diasTranscurridos / duracion) * 100));

            let etapaNombre = "";
            let etapaNumero = 1;
            let colorHex = "#10b981";
            let colorStatus = "verde";
            let estadoTexto = "EN PROCESO";

            if (porcentajeAvance >= 100) {
                etapaNombre = "Cosecha Próxima / Lista";
                etapaNumero = 4;
                colorHex = "#eab308"; // Amarillo Cosecha Próxima
                colorStatus = "amarillo";
                estadoTexto = "Cosecha Próxima / Lista";
            } else if (porcentajeAvance <= 20) {
                etapaNombre = "Siembra / Germinación";
                etapaNumero = 1;
                colorHex = "#f97316"; // Naranja Preparación / Siembra
                colorStatus = "naranja";
                estadoTexto = "Germinación";
            } else {
                etapaNombre = "Desarrollo Vegetativo";
                etapaNumero = 2;
                colorHex = "#10b981"; // Verde Vegetativo Activo
                colorStatus = "verde";
                estadoTexto = "Vegetativo";
            }

            return {
                diasTranscurridos: diasTranscurridos,
                duracionEstimadaDias: duracion,
                porcentajeAvance: porcentajeAvance,
                etapaNombre: etapaNombre,
                etapaNumero: etapaNumero,
                colorHex: colorHex,
                colorStatus: colorStatus,
                estadoTexto: estadoTexto
            };
        }

        function compressImageFile(file, maxWidth = 1024, quality = 0.70) {
            return new Promise((resolve, reject) => {
                const reader = new FileReader();
                reader.onload = function(e) {
                    const img = new Image();
                    img.onload = function() {
                        let width = img.width;
                        let height = img.height;

                        if (width > maxWidth || height > maxWidth) {
                            if (width > height) {
                                height = Math.round((height * maxWidth) / width);
                                width = maxWidth;
                            } else {
                                width = Math.round((width * maxWidth) / height);
                                height = maxWidth;
                            }
                        }

                        const canvas = document.createElement('canvas');
                        canvas.width = width;
                        canvas.height = height;
                        const ctx = canvas.getContext('2d');
                        ctx.drawImage(img, 0, 0, width, height);

                        const compressedBase64 = canvas.toDataURL('image/jpeg', quality);
                        resolve(compressedBase64);
                    };
                    img.onerror = (err) => reject(err);
                    img.src = e.target.result;
                };
                reader.onerror = (err) => reject(err);
                reader.readAsDataURL(file);
            });
        }

                    Titulo: titulo,
                    Descripcion: desc,
                    Base64_Data: base64Data,
                    Url_Foto_Drive: ""
                };

                // Add to local memory immediately for preview
                if (!rawData) rawData = {};
                if (!rawData.fotosLotes) rawData.fotosLotes = [];
                rawData.fotosLotes.push(record);
                renderPhotoGallery();
                document.getElementById("photoUploadForm").reset();
                if (document.getElementById("uploadPhotoFecha")) {
                    document.getElementById("uploadPhotoFecha").value = new Date().toISOString().split('T')[0];
                }

                if (!navigator.onLine) {
                    saveOfflineRecord("fotosLotes", record).then(() => {
                        alert("¡Información registrada con éxito! (Guardado localmente sin conexión)");
                    });
                    return;
                }

                if (typeof google === 'undefined' || !google.script || !google.script.run) {
                    alert("¡Fotografía guardada con éxito!");
                } else {
                    google.script.run
                        .withSuccessHandler(function(response) {
                            if (response && response.success) {
                                if (response.urlDrive) {
                                    record.Url_Foto_Drive = response.urlDrive;
                                    renderPhotoGallery();
                                }
                                alert("¡Fotografía guardada y subida a Google Drive con éxito!");
                            } else {
                                saveOfflineRecord("fotosLotes", record).then(() => {
                                    alert("¡Fotografía guardada localmente!");
                                });
                            }
                        })
                        .withFailureHandler(function(err) {
                            saveOfflineRecord("fotosLotes", record).then(() => {
                                alert("¡Fotografía guardada localmente sin conexión!");
                            });
                        })
                        .guardarFotoLote(record);
                }
            }).catch(err => {
                console.error("Error al comprimir la fotografía:", err);
                alert("Ocurrió un error al procesar la fotografía.");
            });
        }

        function getCurrentGalleryPhotos() {
            const filterSelect = document.getElementById("galleryLoteFilter");
            const selectedLoteFilter = filterSelect ? filterSelect.value : "Todos";
            let photos = (rawData && rawData.fotosLotes) ? rawData.fotosLotes : [];

            if (selectedLoteFilter && selectedLoteFilter !== "Todos") {
                photos = photos.filter(f => String(f.Lote || f.lote || '').trim() === selectedLoteFilter);

            photos.sort((a, b) => String(b.Fecha || b.fecha || '').localeCompare(String(a.Fecha || a.fecha || '')));

        function updateLightboxTransform() {
            const img = document.getElementById("lightboxImage");
            const text = document.getElementById("lightboxZoomLevelText");
            if (!img) return;

            const maxPan = (lightboxZoomScale - 1) * 250;
            lightboxPanX = Math.max(-maxPan, Math.min(maxPan, lightboxPanX));
            lightboxPanY = Math.max(-maxPan, Math.min(maxPan, lightboxPanY));

            img.style.transform = `scale(${lightboxZoomScale}) translate(${lightboxPanX / lightboxZoomScale}px, ${lightboxPanY / lightboxZoomScale}px)`;
            if (text) text.textContent = `${Math.round(lightboxZoomScale * 100)}%`;
        }

        function lightboxZoomIn() {
            lightboxZoomScale = Math.min(3.0, lightboxZoomScale + 0.35);
            updateLightboxTransform();
        }

        function lightboxZoomOut() {
            lightboxZoomScale = Math.max(1.0, lightboxZoomScale - 0.35);
            if (lightboxZoomScale === 1.0) {
                lightboxPanX = 0;
                lightboxPanY = 0;
            }
            updateLightboxTransform();
        }

        function resetLightboxZoom() {
            lightboxZoomScale = 1.0;
            lightboxPanX = 0;
            lightboxPanY = 0;
            updateLightboxTransform();
        }

        function lightboxPrevPhoto() {
            if (lightboxPhotosList.length <= 1) return;
            currentLightboxIndex = (currentLightboxIndex - 1 + lightboxPhotosList.length) % lightboxPhotosList.length;
            openPhotoLightbox(currentLightboxIndex);
        }

        function lightboxNextPhoto() {
            if (lightboxPhotosList.length <= 1) return;
            currentLightboxIndex = (currentLightboxIndex + 1) % lightboxPhotosList.length;
            openPhotoLightbox(currentLightboxIndex);
        }

        function setupLightboxInteractions() {
            const viewport = document.getElementById("lightboxViewport");
            if (!viewport) return;

            viewport.addEventListener("wheel", (e) => {
                const modal = document.getElementById("photoLightboxModal");
                if (modal && !modal.classList.contains("hidden")) {
                    e.preventDefault();
                    if (e.deltaY < 0) {
                        lightboxZoomIn();
                    } else {
                        lightboxZoomOut();
                    }
                }
            }, { passive: false });

            viewport.addEventListener("mousedown", (e) => {
                if (lightboxZoomScale > 1) {
                    isDraggingImage = true;
                    startDragX = e.clientX - lightboxPanX;
                    startDragY = e.clientY - lightboxPanY;
                }
            });

            window.addEventListener("mousemove", (e) => {
                if (isDraggingImage && lightboxZoomScale > 1) {
                    lightboxPanX = e.clientX - startDragX;
                    lightboxPanY = e.clientY - startDragY;
                    updateLightboxTransform();
                }
            });

            window.addEventListener("mouseup", () => {
                isDraggingImage = false;
            });

            window.addEventListener("keydown", (e) => {
                const modal = document.getElementById("photoLightboxModal");
                if (modal && !modal.classList.contains("hidden")) {
                    if (e.key === "Escape") closePhotoLightbox();
                    else if (e.key === "ArrowLeft") lightboxPrevPhoto();
                    else if (e.key === "ArrowRight") lightboxNextPhoto();
                }
            });
        }

        function checkCurrentUserRole() {
            const urlParams = new URLSearchParams(window.location.search);

                    .getCurrentUserRole(savedEmail);

                currentUserRole = "admin";

        function handleCurrentUserRoleSuccess(response) {
            if (!response) return;

            currentUserRole = (response.role || "view").toLowerCase();
            isUserActive = response.active !== false;
            isUserRegistered = response.registered !== false;

            if (response.email) {
                try { localStorage.setItem("finca_user_email", response.email); } catch(e){}
            }

            if (!isUserRegistered || !isUserActive) {
                showAccessBlockedOverlay(response.email || "");
                return;
            }

            const blockedOverlay = document.getElementById("accessBlockedOverlay");
            if (blockedOverlay) blockedOverlay.classList.add("hidden");

            // Aplicar permisos específicos por módulo si existen en la respuesta
            if (response.permissions) {
                applyModuleSpecificPermissions(response.permissions, currentUserRole);
            } else {
                if (currentUserRole === "admin") {
                    restoreAdminPermissions();
                } else {
                    currentUserRole = "view";
                    applyViewOnlyPermissions();
                }
            }
        }

        function applyModuleSpecificPermissions(perms, role) {
            if (role === "admin") {
                restoreAdminPermissions();
                return;
            }

            const isPermitted = function(val) {
                if (val === undefined || val === null || val === "") return true; // Default to allowed for registered active users if unconfigured
                const s = String(val).trim().toLowerCase();
                return s === "sí" || s === "si" || s === "true" || s === "1" || s === "activo";
            };

            // Ocultar / Mostrar menús y botones según los permisos específicos por módulo
            const modDashboard = isPermitted(perms.Modulo_Dashboard || perms.Modulo_Inicio);
            const modPOS = isPermitted(perms.Modulo_POS || perms.Modulo_Gastos);
            const modEstadoResultados = isPermitted(perms.Modulo_EstadoResultados || perms.Modulo_Finanzas);
            const modObligaciones = isPermitted(perms.Modulo_Obligaciones || perms.Modulo_Finanzas);
            const modMapaLotes = isPermitted(perms.Modulo_MapaLotes || perms.Modulo_Lotes);
            const modCajaGeneral = isPermitted(perms.Modulo_CajaGeneral || perms.Modulo_Finanzas);

            // Modulo Dashboard
            const dashBtns = document.querySelectorAll("#navBtn_dashboard, #mobileNavBtn_dashboard, [data-module='Modulo_Dashboard']");
            dashBtns.forEach(b => b.style.display = modDashboard ? "" : "none");

            // Modulo POS / Ingresar Info
            const posBtns = document.querySelectorAll("#toggleViewBtn, #mobileToggleViewBtn, [data-module='Modulo_POS']");
            posBtns.forEach(b => b.style.display = modPOS ? "" : "none");

            // Modulo Estado Resultados
            const erBtns = document.querySelectorAll("#estadoResultadosBtn, #mobileEstadoResultadosBtn, [data-module='Modulo_EstadoResultados']");
            erBtns.forEach(b => b.style.display = modEstadoResultados ? "" : "none");

            // Modulo Obligaciones
            const obBtns = document.querySelectorAll("#obligacionesBtn, #mobileObligacionesBtn, [data-module='Modulo_Obligaciones']");
            obBtns.forEach(b => b.style.display = modObligaciones ? "" : "none");

            if (typeof google === 'undefined' || !google.script || !google.script.run) {
                const u = usuariosListCache.find(x => x.email === email);
                if (u) u.estado = nuevoEstado;
                renderUsersTable();
                showLoader(false);
            } else {
                google.script.run
                    .withSuccessHandler(function(res) {
                        showLoader(false);
                        if (res && res.success) {
                            fetchUsersList();
                        } else {
                            alert(res ? res.error : "No se pudo cambiar el estado.");
                        }
                    })
                    .withFailureHandler(function(err) {
                        showLoader(false);
                        alert("Error: " + err.toString());
                    })
                    .guardarUsuario(record);
            }
        }

        function deleteUser(email) {
            if (!confirm(`¿Estás seguro de eliminar al usuario ${email}? No podrá volver a ingresar.`)) return;

            showLoader(true, "Eliminando usuario...");

            if (typeof google === 'undefined' || !google.script || !google.script.run) {
                usuariosListCache = usuariosListCache.filter(u => u.email !== email);
                renderUsersTable();
                showLoader(false);
            } else {
                google.script.run
                    .withSuccessHandler(function(res) {
                        showLoader(false);
                        if (res && res.success) {
                            fetchUsersList();
                        } else {
                            alert(res ? res.error : "No se pudo eliminar el usuario.");
                        }
                    })
                    .withFailureHandler(function(err) {
                        showLoader(false);
                        alert("Error: " + err.toString());
                    })
                    .eliminarUsuario(email);
            }
        }

        function applyViewOnlyPermissions() {
            if (currentUserRole !== 'view') return;

            // Disable or hide creation/edit/delete buttons
            const buttonsToHide = document.querySelectorAll(
                "#toggleViewBtn, #mobileToggleViewBtn, #obligacionesBtn, #mobileObligacionesBtn, #cajaGeneralBtn, #mobileCajaGeneralBtn, #gestionUsuariosBtn, #mobileUsuariosBtn, #mobileSyncBtn, button[onclick*='openDeleteModal'], button[onclick*='eliminarFotoGallery'], button[onclick*='openPhotoUploadModal'], button[onclick*='marcarComoPagado'], button[onclick*='startLoteCycle'], button[onclick*='endLoteCycle'], button[onclick*=\"showView('data_entry')\"]"
            );
            buttonsToHide.forEach(btn => {
                btn.style.display = "none";
            });

            // Put data entry forms in readonly / disabled mode while preserving filter selects/inputs
            const dataForms = document.querySelectorAll("#posForm, #obligacionesForm, #gastoPersonalForm, #securityChangeForm, #clearDataForm, #formUsuarioAdmin, #photoUploadForm");
            dataForms.forEach(form => {
                const elements = form.querySelectorAll("input, select, textarea, button[type='submit']");
                elements.forEach(el => {
                    if (el.tagName === "BUTTON") {
                        el.disabled = true;
                        el.classList.add("opacity-50", "cursor-not-allowed");
                    } else if (el.tagName === "INPUT" || el.tagName === "TEXTAREA") {
                        el.readOnly = true;
                        el.disabled = true;
                    } else if (el.tagName === "SELECT") {
                        el.disabled = true;
                    }
                });
            });

            // Enable navigation & filter dropdowns explicitly for view users
            const filtersToEnable = document.querySelectorAll("#filterLote, #filterCultivo, #filterCiclo, #loteSelector, #variationItemSelector, #galleryLoteFilter, #filterMovimiento, #searchTrackingCode, #filterFechaDesde, #filterFechaHasta");
            filtersToEnable.forEach(el => {
                if (el) {
                    el.disabled = false;
                    el.readOnly = false;
                }
            });

            // Observer or periodic re-apply to handle dynamically rendered action/navigation buttons
            if (!window.viewOnlyObserverActive) {
                window.viewOnlyObserverActive = true;
                const observer = new MutationObserver(() => {
                    if (currentUserRole === 'view') {
                        const dynamicBtns = document.querySelectorAll(
                            "#toggleViewBtn, #mobileToggleViewBtn, #obligacionesBtn, #mobileObligacionesBtn, #cajaGeneralBtn, #mobileCajaGeneralBtn, #gestionUsuariosBtn, #mobileUsuariosBtn, #mobileSyncBtn, button[onclick*='openDeleteModal'], button[onclick*='eliminarFotoGallery'], button[onclick*='openPhotoUploadModal'], button[onclick*='marcarComoPagado'], button[onclick*='startLoteCycle'], button[onclick*='endLoteCycle'], button[onclick*=\"showView('data_entry')\"]"
                        );
                        dynamicBtns.forEach(btn => {
                            btn.style.display = "none";
                        });
                    }
                });
                observer.observe(document.body, { childList: true, subtree: true });
            }
        }

        // --- MODAL DETALLE DE CONTADOR Y CONSOLIDADOR POR REFERENCIA ---
        function openCounterDetailModal(tipoCategoria) {
            if (!rawData) return;

            const modal = document.getElementById("modalDetalleContador");
            if (!modal) return;

            const selectedLote = document.getElementById("filterLote")?.value || "Todos";
            const selectedCultivo = document.getElementById("filterCultivo")?.value || "Todos";
            const selectedCiclo = document.getElementById("filterCiclo") ? document.getElementById("filterCiclo").value : "Todos";
            const fechaDesde = document.getElementById("filterFechaDesde")?.value || "";
            const fechaHasta = document.getElementById("filterFechaHasta")?.value || "";

            const isDateInRange = (dateStr) => {
                if (!dateStr || dateStr === "-") return true;
                const d = dateStr.trim();
                if (fechaDesde && d < fechaDesde) return false;
                if (fechaHasta && d > fechaHasta) return false;
                return true;
            };

            const isCicloMatch = (item) => {
                if (!selectedCiclo || selectedCiclo === "Todos") return true;
                const code = String(item.Codigo_Seguimiento || item.codigo_seguimiento || item.Codigo_General || item.codigo_general || "").trim().toUpperCase();
                return code === selectedCiclo.trim().toUpperCase();
            };

            const isCultivoMatch = (item) => {
                if (selectedCultivo === "Todos") return true;
                const c = getCultivoForTx(item);
                return c.toLowerCase().indexOf(selectedCultivo.toLowerCase()) !== -1;
            };

            let titleText = "Detalle de Registros";
            let iconSymbol = "📦";
            let rawKey = "gastosIniciales";
            let isVenta = false;

            const catUpper = String(tipoCategoria || "").toLowerCase();
            if (catUpper.includes("inicial")) {
                titleText = "Registros Semilla / Iniciales";
                iconSymbol = "📦";
                rawKey = "gastosIniciales";
            } else if (catUpper.includes("desarrollo")) {
                titleText = "Registros Gastos de Desarrollo";
                iconSymbol = "🌿";
                rawKey = "gastosDesarrollo";
            } else if (catUpper.includes("venta")) {
                titleText = "Registros de Ventas";
                iconSymbol = "💰";
                rawKey = "ventas";
                isVenta = true;
            }

            // Set modal header
            document.getElementById("modalContadorTitulo").textContent = titleText;
            document.getElementById("modalContadorIcon").textContent = iconSymbol;

            let filterSubtext = `Filtros: Lote ${selectedLote} | Cultivo ${selectedCultivo} | Ciclo ${selectedCiclo}`;
            if (fechaDesde || fechaHasta) {
                filterSubtext += ` | Periodo: ${fechaDesde || 'Inicio'} a ${fechaHasta || 'Hoy'}`;
            }
            document.getElementById("modalContadorSubtitulo").textContent = filterSubtext;

            // Filter raw data
            const matchingRecords = (rawData[rawKey] || []).filter(item => {
                return isLoteMatch(item.Lote || item.lote, selectedLote) &&
                       isCicloMatch(item) &&
                       isDateInRange(item.Fecha || item.fecha) &&
                       isCultivoMatch(item);
            });

            // Populate Consolidated Table by Concepto / Reference
            const consolidadoTableBody = document.getElementById("counterConsolidadoTableBody");
            consolidadoTableBody.innerHTML = "";

            const consolidadoMap = {};
            let grandTotalCantidad = 0;
            let grandTotalMonto = 0;

            matchingRecords.forEach(item => {
                let concepto = "";
                let cantidad = 0;
                let monto = 0;

                if (rawKey === "gastosIniciales") {
                    concepto = String(item.Cultivo_Semilla || item.cultivo_semilla || "Semilla / Inicial").trim();
                    cantidad = Number(item.Cantidad || item.cantidad) || 1;
                    monto = Number(item.Costo_Total_COP || item.costo_total_cop) || 0;
                } else if (rawKey === "gastosDesarrollo") {
                    concepto = String(item.Tipo_Insumo || item.tipo_insumo || "Insumo / Gasto").trim();
                    cantidad = Number(item.Cantidad || item.cantidad) || 1;
                    monto = Number(item.Costo_Total_COP || item.costo_total_cop) || 0;
                } else if (rawKey === "ventas") {
                    concepto = String(item.Producto_Vendido || item.producto_vendido || "Producto").trim();
                    cantidad = Number(item.Cantidad || item.cantidad || item.Unidades_Cosechadas || item.unidades_cosechadas) || 1;
                    monto = Number(item.Ingreso_Total_COP || item.ingreso_total_cop) || 0;
                }

                if (!consolidadoMap[concepto]) {
                    consolidadoMap[concepto] = {
                        concepto: concepto,
                        count: 0,
                        cantidadAcumulada: 0,
                        montoTotal: 0
                    };
                }

                consolidadoMap[concepto].count += 1;
                consolidadoMap[concepto].cantidadAcumulada += cantidad;
                consolidadoMap[concepto].montoTotal += monto;

                grandTotalCantidad += cantidad;
                grandTotalMonto += monto;
            });

            const consolidadoItems = Object.values(consolidadoMap).sort((a, b) => b.montoTotal - a.montoTotal);

            if (consolidadoItems.length === 0) {
                consolidadoTableBody.innerHTML = `
                    <tr>
                        <td colspan="4" class="px-4 py-4 text-center text-xs text-slate-400 italic font-semibold">
                            No existen registros consolidados para la categoría seleccionada bajo los filtros actuales.
                        </td>
                    </tr>
                `;
            } else {
                consolidadoItems.forEach(group => {
                    const tr = document.createElement("tr");
                    tr.className = "hover:bg-slate-50 dark:hover:bg-slate-800 transition duration-150";
                    tr.innerHTML = `
                        <td class="px-4 py-2.5 text-left font-bold text-slate-800 dark:text-slate-200">${group.concepto}</td>
                        <td class="px-4 py-2.5 text-center font-semibold text-slate-600 dark:text-slate-400">${group.count}</td>
                        <td class="px-4 py-2.5 text-right font-medium text-slate-700 dark:text-slate-300">${group.cantidadAcumulada.toLocaleString('es-CO')}</td>
                        <td class="px-4 py-2.5 text-right font-extrabold ${isVenta ? 'text-emerald-600 dark:text-emerald-400' : 'text-slate-900 dark:text-white'}">${formatCOP(group.montoTotal)}</td>
                    `;
                    consolidadoTableBody.appendChild(tr);
                });

                // Total Summary Row
                const totalTr = document.createElement("tr");
                totalTr.className = "bg-slate-100 dark:bg-slate-800/80 font-black border-t-2 border-slate-200 dark:border-slate-700";
                totalTr.innerHTML = `
                    <td class="px-4 py-2.5 text-left uppercase text-slate-700 dark:text-slate-200">TOTAL ACUMULADO</td>
                    <td class="px-4 py-2.5 text-center text-slate-800 dark:text-slate-200">${matchingRecords.length}</td>
                    <td class="px-4 py-2.5 text-right text-slate-800 dark:text-slate-200">${grandTotalCantidad.toLocaleString('es-CO')}</td>
                    <td class="px-4 py-2.5 text-right font-black ${isVenta ? 'text-emerald-600 dark:text-emerald-400' : 'text-slate-900 dark:text-white'}">${formatCOP(grandTotalMonto)}</td>
                `;
                consolidadoTableBody.appendChild(totalTr);
            }

            // Populate Itemized Breakdown Table
            const desgloseTableBody = document.getElementById("counterDesgloseTableBody");
            desgloseTableBody.innerHTML = "";

            // Sort by date descending
            matchingRecords.sort((a, b) => {
                const fechaA = a.Fecha || a.fecha || "";
                const fechaB = b.Fecha || b.fecha || "";
                return fechaB.localeCompare(fechaA);
            });

            if (matchingRecords.length === 0) {
                desgloseTableBody.innerHTML = `
                    <tr>
                        <td colspan="7" class="px-4 py-4 text-center text-xs text-slate-400 italic font-semibold">
                            Sin transacciones registradas bajo estos parámetros.
                        </td>
                    </tr>
                `;
            } else {
                matchingRecords.forEach(item => {
                    const fecha = item.Fecha || item.fecha || "-";
                    const lote = item.Lote || item.lote || "-";
                    const cultivo = getCultivoForTx(item);

                    let concepto = "";
                    let cantidad = 1;
                    let unidad = "Unidades";
                    let monto = 0;

                    if (rawKey === "gastosIniciales") {
                        concepto = item.Cultivo_Semilla || item.cultivo_semilla || "Semilla/Inicial";
                        cantidad = Number(item.Cantidad || item.cantidad) || 1;
                        unidad = item.Unidad_Medida || item.unidad_medida || "Unidades";
                        monto = Number(item.Costo_Total_COP || item.costo_total_cop) || 0;
                    } else if (rawKey === "gastosDesarrollo") {
                        concepto = item.Tipo_Insumo || item.tipo_insumo || "Insumo/Gasto";
                        cantidad = Number(item.Cantidad || item.cantidad) || 1;
                        unidad = item.Unidad_Medida || item.unidad_medida || "Unidades";
                        monto = Number(item.Costo_Total_COP || item.costo_total_cop) || 0;
                    } else if (rawKey === "ventas") {
                        concepto = item.Producto_Vendido || item.producto_vendido || "Producto";
                        cantidad = Number(item.Cantidad || item.cantidad || item.Unidades_Cosechadas || item.unidades_cosechadas) || 1;
                        unidad = item.Unidad_Medida || item.unidad_medida || "Atados";
                        monto = Number(item.Ingreso_Total_COP || item.ingreso_total_cop) || 0;
                    }

                    const tr = document.createElement("tr");
                    tr.className = "hover:bg-slate-50 dark:hover:bg-slate-800 transition duration-150";
                    tr.innerHTML = `
                        <td class="px-4 py-2.5 text-left text-slate-500 font-medium font-mono">${fecha}</td>
                        <td class="px-4 py-2.5 text-left font-bold text-slate-900 dark:text-white">${lote}</td>
                        <td class="px-4 py-2.5 text-left text-slate-600 dark:text-slate-300 font-medium">${cultivo}</td>
                        <td class="px-4 py-2.5 text-left font-semibold text-slate-800 dark:text-slate-200">${concepto}</td>
                        <td class="px-4 py-2.5 text-right font-medium text-slate-700 dark:text-slate-300">${cantidad.toLocaleString('es-CO')}</td>
                        <td class="px-4 py-2.5 text-center text-slate-500 text-[11px]">${unidad}</td>
                        <td class="px-4 py-2.5 text-right font-bold ${isVenta ? 'text-emerald-600 dark:text-emerald-400' : 'text-slate-800 dark:text-slate-100'}">${formatCOP(monto)}</td>
                    `;
                    desgloseTableBody.appendChild(tr);
                });
            }

            modal.classList.remove("hidden");
        }
