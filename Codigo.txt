function generarCodigoSeguimiento(loteCompleto, cultivo, fecha) {
  var lClean = String(loteCompleto || '').toUpperCase().replace(/[^A-Z0-9]/g, '');
  var cClean = String(cultivo || '').toUpperCase().replace(/[^A-Z0-9]/g, '');
  var yClean = '2026';
  if (fecha) {
    var parts = String(fecha).split('-');
    if (parts.length >= 1 && parts[0].length === 4) yClean = parts[0];
  }
  if (!lClean) lClean = 'LOTE';
  if (!cClean) cClean = 'CULTIVO';
  return 'INI-' + lClean + '-' + cClean + '-' + yClean;
}

function calcularEtapaCultivo(fechaInicio, duracionDias, cultivo) {
  var duracion = Number(duracionDias) || 0;
  if (!duracion) {
    var cultLower = String(cultivo || '').toLowerCase();
    if (cultLower.indexOf('cilantro') !== -1) duracion = 60;
    else if (cultLower.indexOf('arveja') !== -1) duracion = 120;
    else if (cultLower.indexOf('cebolla cabezona') !== -1) duracion = 140;
    else if (cultLower.indexOf('cebolla') !== -1) duracion = 150;
    else if (cultLower.indexOf('café') !== -1 || cultLower.indexOf('cafe') !== -1) duracion = 300;
    else if (cultLower.indexOf('aguacate') !== -1) duracion = 365;
    else if (cultLower.indexOf('cacao') !== -1) duracion = 300;
    else if (cultLower.indexOf('plátano') !== -1 || cultLower.indexOf('platano') !== -1) duracion = 270;
    else duracion = 150;
  }

  if (!fechaInicio || fechaInicio === "-" || fechaInicio === "") {
    return {
      diasTranscurridos: 0,
      duracionEstimadaDias: duracion,
      porcentajeAvance: 0,
      etapaNombre: "Sin Ciclo Activo (Barbecho / Descanso)",
      etapaNumero: 0,
      colorHex: "#64748b",
      colorStatus: "gris",
      estadoTexto: "LIBRE"
    };
  }

  var hoy = new Date();
  var inicio = new Date(fechaInicio);
  var diffTime = Math.max(0, hoy.getTime() - inicio.getTime());
  var diasTranscurridos = Math.floor(diffTime / (1000 * 60 * 60 * 24));
  var porcentajeAvance = Math.min(100, Math.round((diasTranscurridos / duracion) * 100));

  var etapaNombre = "";
  var etapaNumero = 1;
  var colorHex = "#10b981";
  var colorStatus = "verde";

  if (porcentajeAvance <= 20) {
    etapaNombre = "Etapa 1: Germinación y Arraigo / Preparación";
    etapaNumero = 1;
    colorHex = "#f97316";
    colorStatus = "naranja";
  } else if (porcentajeAvance <= 60) {
    etapaNombre = "Etapa 2: Desarrollo Vegetativo y Crecimiento";
    etapaNumero = 2;
    colorHex = "#10b981";
    colorStatus = "verde";
  } else if (porcentajeAvance <= 85) {
    etapaNombre = "Etapa 3: Llenado de Cabezas / Bulbo / Floración";
    etapaNumero = 3;
    colorHex = "#10b981";
    colorStatus = "verde";
  } else {
    etapaNombre = "Etapa 4: Cosecha y Recolección";
    etapaNumero = 4;
    colorHex = "#eab308";
    colorStatus = "amarillo";
  }

  return {
    diasTranscurridos: diasTranscurridos,
    duracionEstimadaDias: duracion,
    porcentajeAvance: porcentajeAvance,
    etapaNombre: etapaNombre,
    etapaNumero: etapaNumero,
    colorHex: colorHex,
    colorStatus: colorStatus,
    estadoTexto: "EN PROCESO"
  };
}

/**
 * Función para crear un backup diario automático del documento actual
 * en la carpeta especificada de Google Drive (ID: 1B7Um_1m_dcuq9r85qV6TcW1LG0QXnRzu).
 */
function crearBackupDiario() {
  return runDailyBackup();
}

/**
 * Función de Respaldo Diario Automatizado (runDailyBackup)
 * Captura el estado actual de las fuentes de datos (Google Sheets / JSON consolidado),
 * genera un respaldo espejo con marca de tiempo (YYYY-MM-DD) y lo almacena en Google Drive.
 */
function runDailyBackup() {
  try {
    var folderId = "1B7Um_1m_dcuq9r85qV6TcW1LG0QXnRzu";
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var timeZone = ss.getSpreadsheetTimeZone() || "America/Bogota";
    var dateFormatted = Utilities.formatDate(new Date(), timeZone, "yyyy-MM-dd_HH-mm");
    var backupName = "Backup_" + ss.getName() + "_" + dateFormatted;

    var folder;
    try {
      folder = DriveApp.getFolderById(folderId);
    } catch (fErr) {
      // Si la carpeta designada no existe o no tiene permisos, crear carpeta por defecto en Drive
      var folderName = "Finca Los Sauces - Backups Automatizados";
      var folders = DriveApp.getFoldersByName(folderName);
      folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
    }

    // 1. Generar copia espejo del libro de cálculo
    var file = DriveApp.getFileById(ss.getId());
    var backupFile = file.makeCopy(backupName, folder);

    // 2. Generar respaldo estructurado consolidado en formato JSON
    var consolidatedData = getDashboardData();
    if (consolidatedData && consolidatedData.success) {
      var jsonContent = JSON.stringify(consolidatedData.data, null, 2);
      var jsonBlob = Utilities.newBlob(jsonContent, "application/json", backupName + ".json");
      folder.createFile(jsonBlob);
    }

    Logger.log("Backup diario ejecutado con éxito: " + backupName);
    return { success: true, backupName: backupName, fileUrl: backupFile.getUrl() };
  } catch (err) {
    Logger.log("ERROR CRÍTICO EN BACKUP DIARIO: " + err.toString());
    return { success: false, error: err.toString() };
  }
}

/**
 * Disparador Automático (setupDailyTrigger / configurarActivadorBackupDiario)
 * Programa automáticamente el activador time-driven diariamente a las 00:00 h.
 */
function setupDailyTrigger() {
  try {
    var triggers = ScriptApp.getProjectTriggers();
    for (var i = 0; i < triggers.length; i++) {
      var funcName = triggers[i].getHandlerFunction();
      if (funcName === "runDailyBackup" || funcName === "crearBackupDiario") {
        ScriptApp.deleteTrigger(triggers[i]);
      }
    }

    ScriptApp.newTrigger("runDailyBackup")
      .timeBased()
      .everyDays(1)
      .atHour(0)
      .create();

    Logger.log("Activador de backup diario a las 00:00 h configurado correctamente.");
    return { success: true, message: "Activador automático diario configurado a medianoche." };
  } catch (err) {
    Logger.log("Error al configurar activador diario: " + err.toString());
    return { success: false, error: err.toString() };
  }
}

function configurarActivadorBackupDiario() {
  return setupDailyTrigger();
}

          if (mimeMatch && mimeMatch[1]) mimeType = mimeMatch[1];
        }

        var bytes = Utilities.base64Decode(base64Clean);
        var blob = Utilities.newBlob(bytes, mimeType, idFoto + "_" + lote.replace(/\s+/g, "") + ".jpg");
        var driveFile = targetFolder.createFile(blob);
        driveFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        urlDrive = driveFile.getUrl();
      } catch (driveErr) {
        Logger.log("Error subiendo foto a Drive: " + driveErr.toString());
      }
    }

    sheet.appendRow([idFoto, lote, fecha, titulo, descripcion, urlDrive, base64Data]);
    return { success: true, idFoto: idFoto, urlDrive: urlDrive, message: "Fotografía guardada con éxito." };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

    var ventas = getSheetData(ss, HOJA_VENTAS);
    var fotos = getSheetData(ss, HOJA_FOTOS_LOTES);

    var lotesMap = {};

    for (var m = 1; m <= 16; m++) {
      var loteName = "Lote " + m;

      var activeCycle = ciclos.find(function(c) {
        var cLote = String(c.Lote || c.lote || c.Lote_Principal || '').trim();
        return (cLote === loteName || cLote.indexOf(loteName + " ") === 0) && String(c.Estado || c.estado || '').toUpperCase() === 'EN PROCESO';

      var iniRow = gIniciales.find(function(ini) {
        var iLote = String(ini.Lote || ini.lote || ini.Lote_Principal || '').trim();
        return iLote === loteName || iLote.indexOf(loteName + " ") === 0;

      var cultivoName = "Barbecho / Descanso";

    var fechaInicio = datos.Fecha_Inicio || datos.fecha_inicio || formatDateNative(new Date());

    var resCiclo = agregarCiclo(lp, sub, cultivo, trackingCode, trackingCode, trackingCode, trackingCode, "EN PROCESO", fechaInicio, "");

      });
    }

    return {
      success: true,
      lote: loteCompleto,
      codigoGeneral: trackingCode,
      message: "¡Lote " + loteCompleto + " activado y vinculado dinámicamente al mapa interactivo!"
    };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function obtenerDatosConsolidados(filtros) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = {
      gastosIniciales: getSheetData(ss, HOJA_GASTOS_INICIALES),
      gastosDesarrollo: getSheetData(ss, HOJA_GASTOS_DESARROLLO),
      ventas: getSheetData(ss, HOJA_VENTAS),
      ciclos: getSheetData(ss, HOJA_CICLOS),
      prestamos: getSheetData(ss, HOJA_PRESTAMOS)
    };

    var fechaInicio = filtros ? filtros.fechaInicio : null;
    var fechaFin = filtros ? filtros.fechaFin : null;
    var lote = filtros ? filtros.lote : "Todos";
    var cultivo = filtros ? filtros.cultivo : "Todos";

    var filterFn = function(item, type) {
      var itemFecha = item.Fecha || item.fecha || "";
      var itemLote = item.Lote || item.lote || item.Lote_Completo || "";
      var itemMain = item.Lote_Principal || item.lote_principal || "";

      if (fechaInicio && itemFecha < fechaInicio) return false;
      if (fechaFin && itemFecha > fechaFin) return false;

      if (lote && lote !== "Todos") {
        var cleanLote = lote.replace(/\(Consolidado.*?\)/i, "").trim();
        var isConsolidated = lote.indexOf("Consolidado") !== -1 || cleanLote.match(/^Lote\s*\d+$/i);

        if (isConsolidated) {
          var matchMain = (itemMain === cleanLote) ||
                          (itemLote === cleanLote) ||
                          (itemLote.indexOf(cleanLote + " ") === 0) ||
                          (itemLote.indexOf(cleanLote) === 0 && (itemLote.length === cleanLote.length || itemLote.charAt(cleanLote.length).match(/[A-Z0-9]/i)));
          if (!matchMain) return false;
        } else {
          if (itemLote !== cleanLote && itemLote !== lote) return false;
        }
      }

      if (cultivo && cultivo !== "Todos") {
        var itemCultivo = "";
        if (type === "inicial") itemCultivo = item.Cultivo_Semilla || item.cultivo_semilla || "";
        else if (type === "venta") itemCultivo = item.Producto_Vendido || item.producto_vendido || "";

        if (!itemCultivo && data.ciclos) {
          var trackingCode = (item.Codigo_General || item.Codigo_Seguimiento || "").toString().trim().toUpperCase();
          if (trackingCode) {
            var cycle = data.ciclos.find(function(c) {
              return (c.Codigo_General || "").toString().trim().toUpperCase() === trackingCode ||
                     (c.Codigo_Inicial || "").toString().trim().toUpperCase() === trackingCode ||
                     (c.Codigo_Desarrollo || "").toString().trim().toUpperCase() === trackingCode ||
                     (c.Codigo_Ventas || "").toString().trim().toUpperCase() === trackingCode;
            });
            if (cycle) itemCultivo = cycle.Cultivo || cycle.cultivo || "";
          }
        }

        if (!itemCultivo && data.gastosIniciales) {
          var firstIni = data.gastosIniciales.find(function(ini) {
            return (ini.Lote || "").toString().trim() === itemLote;
          });
          if (firstIni) itemCultivo = firstIni.Cultivo_Semilla || firstIni.cultivo_semilla || "";
        }

        if (itemCultivo.toLowerCase().indexOf(cultivo.toLowerCase()) === -1) return false;
      }
      return true;
    };

    var filteredIniciales = data.gastosIniciales.filter(function(x) { return filterFn(x, "inicial"); });
    var filteredDesarrollo = data.gastosDesarrollo.filter(function(x) { return filterFn(x, "desarrollo"); });
    var filteredVentas = data.ventas.filter(function(x) { return filterFn(x, "venta"); });

    // Cash flow & payment status calculations
    var cajaReal = filteredVentas
      .filter(function(x) { return String(x.Estado_Pago || x.estado_pago || 'PAGADO').toUpperCase() === 'PAGADO'; })
      .reduce(function(sum, x) { return sum + (Number(x.Ingreso_Total_COP || x.ingreso_total_cop) || 0); }, 0);

    var salidaReal = filteredIniciales.concat(filteredDesarrollo)
      .filter(function(x) { return String(x.Estado_Pago || x.estado_pago || 'PAGADO').toUpperCase() === 'PAGADO'; })
      .reduce(function(sum, x) { return sum + (Number(x.Costo_Total_COP || x.costo_total_cop) || 0); }, 0);

    var cuentasPorCobrar = filteredVentas
      .filter(function(x) { return String(x.Estado_Pago || x.estado_pago || '').toUpperCase() === 'PENDIENTE'; })
      .reduce(function(sum, x) { return sum + (Number(x.Ingreso_Total_COP || x.ingreso_total_cop) || 0); }, 0);

    var cuentasPorPagar = filteredIniciales.concat(filteredDesarrollo)
      .filter(function(x) { return String(x.Estado_Pago || x.estado_pago || '').toUpperCase() === 'PENDIENTE'; })
      .reduce(function(sum, x) { return sum + (Number(x.Costo_Total_COP || x.costo_total_cop) || 0); }, 0);

    var saldoDisponible = cajaReal - salidaReal;
    var saldoProyectado = (cajaReal + cuentasPorCobrar) - cuentasPorPagar;

    var totalGastos = filteredIniciales.concat(filteredDesarrollo)
      .reduce(function(sum, x) { return sum + (Number(x.Costo_Total_COP || x.costo_total_cop) || 0); }, 0);

    var totalVentas = filteredVentas
      .reduce(function(sum, x) { return sum + (Number(x.Ingreso_Total_COP || x.ingreso_total_cop) || 0); }, 0);

    var totalUnidades = filteredVentas
      .reduce(function(sum, x) { return sum + (Number(x.Cantidad || x.cantidad || x.Unidades_Cosechadas || x.unidades_cosechadas) || 0); }, 0);

    var utilidadNeta = totalVentas - totalGastos;
    var roi = totalGastos > 0 ? (utilidadNeta / totalGastos) * 100 : 0;
    var costoUnitario = totalUnidades > 0 ? (totalGastos / totalUnidades) : 0;
    var precioVentaPromedio = totalUnidades > 0 ? (totalVentas / totalUnidades) : 0;
    var margenUnitario = precioVentaPromedio - costoUnitario;

    var totalDeudaBank = data.prestamos.reduce(function(sum, x) { return sum + (Number(x.Saldo_Por_Pagar || x.saldo_por_pagar) || 0); }, 0);

    return {
      success: true,
      metrics: {
        cajaReal: cajaReal,
        salidaReal: salidaReal,
        saldoDisponible: saldoDisponible,
        cuentasPorCobrar: cuentasPorCobrar,
        cuentasPorPagar: cuentasPorPagar,
        saldoProyectado: saldoProyectado,
        gastosTotales: totalGastos,
        ingresosTotales: totalVentas,
        utilidadNeta: utilidadNeta,
        roiPorcentaje: roi,
        unidadesCosechadasTotales: totalUnidades,
        costoUnitarioProduccion: costoUnitario,
        precioVentaPromedio: precioVentaPromedio,
        margenUnitario: margenUnitario,
        deudaTotalActiva: totalDeudaBank
      },
      data: {
        gastosIniciales: filteredIniciales,
        gastosDesarrollo: filteredDesarrollo,
        ventas: filteredVentas
      }
    };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function obtenerListaLotesActivos(ss) {
  if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
  var list = [];
  var ciclosSheet = ss.getSheetByName(HOJA_CICLOS);
  
  if (ciclosSheet && ciclosSheet.getLastRow() >= 2) {
    var ciclosData = leerTablaHoja(HOJA_CICLOS);
    ciclosData.forEach(function(c) {
      var estado = String(c.Estado || c.estado || '').toUpperCase();
      if (estado === 'EN PROCESO' || estado === 'ACTIVO') {
        var lp = String(c.Lote_Principal || c.lote_principal || c.Lote || c.lote || '').trim();
        var match = lp.match(/^(Lote\s*\d+)/i);
        var loteName = match ? match[1] : lp;
        if (loteName && list.indexOf(loteName) === -1) {
          list.push(loteName);
        }
      }
    });
  }

  // Fallback if no active cycles registered: default to all 16 lotes
  if (list.length === 0) {
    for (var i = 1; i <= 16; i++) {
      list.push("Lote " + i);
    }
  }

  return list;
}

function guardarRegistro(tipo, datos) {
  try {
    var authCheck = verificarAccesoUsuario();
    if (!authCheck.autorizado) return { success: false, error: authCheck.error || "Acceso No Autorizado." };

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "";
    var columnsNeeded = [];

    if (tipo === "gastosIniciales" || tipo === "inicial") {
      sheetName = HOJA_GASTOS_INICIALES;
      columnsNeeded = ["Lote_Principal", "Sub_Lote", "Lote", "Cultivo_Semilla", "Cantidad", "Unidad_Medida", "Costo_Unitario_COP", "Costo_Total_COP", "Estado_Pago", "Cliente_Deudor_Proveedor", "Fecha_Vencimiento", "Fecha_Pago_Realizado", "Fecha", "Notas", "Codigo_Seguimiento", "Codigo_General"];
    } else if (tipo === "gastosDesarrollo" || tipo === "desarrollo") {
      sheetName = HOJA_GASTOS_DESARROLLO;
      columnsNeeded = ["Lote_Principal", "Sub_Lote", "Lote", "Tipo_Insumo", "Cantidad", "Unidad_Medida", "Costo_Unitario_COP", "Costo_Total_COP", "Estado_Pago", "Cliente_Deudor_Proveedor", "Fecha_Vencimiento", "Fecha_Pago_Realizado", "Fecha", "Detalle", "Codigo_Seguimiento", "Codigo_General"];
    } else if (tipo === "ventas" || tipo === "venta") {
      sheetName = HOJA_VENTAS;
      columnsNeeded = ["Lote_Principal", "Sub_Lote", "Lote", "Producto_Vendido", "Cantidad", "Unidad_Medida", "Precio_Unitario_COP", "Ingreso_Total_COP", "Estado_Pago", "Cliente_Deudor_Proveedor", "Fecha_Vencimiento", "Fecha_Pago_Realizado", "Fecha", "Comprador", "Codigo_Seguimiento", "Codigo_General"];
    } else {
      return { success: false, error: "Tipo de registro no válido." };
    }

    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(columnsNeeded);
    }
    asegurarColumnas(sheet, columnsNeeded);

    var cantidad = Number(datos.Cantidad || datos.cantidad) || 1;
    var unitario = Number(datos.Costo_Unitario_COP || datos.Precio_Unitario_COP || datos.costo_unitario || datos.precio_unitario) || 0;
    var total = Number(datos.Costo_Total_COP || datos.Ingreso_Total_COP || datos.costo_total || datos.ingreso_total) || 0;

    if (total === 0 && unitario > 0) {
      total = cantidad * unitario;
    } else if (unitario === 0 && total > 0 && cantidad > 0) {
      unitario = total / cantidad;
    }

    var estadoPago = (datos.Estado_Pago || datos.estado_pago || "PAGADO").toUpperCase();
    var cliente = datos.Cliente_Deudor_Proveedor || datos.cliente_deudor_proveedor || datos.Comprador || datos.comprador || "";
    var fechaVenc = datos.Fecha_Vencimiento || datos.fecha_vencimiento || "";

    var tipoInsumo = datos.Tipo_Insumo || datos.tipo_insumo || "";
    var isProrrateoInsumo = (tipo === "gastosDesarrollo" || tipo === "desarrollo") && (tipoInsumo === "Fertilizantes" || tipoInsumo === "Pesticidas");

    if (isProrrateoInsumo) {
      var lotesActivos = obtenerListaLotesActivos(ss);
      var numLotes = lotesActivos.length;
      var propMonto = numLotes > 0 ? (total / numLotes) : total;
      var propCantidad = numLotes > 0 ? (cantidad / numLotes) : cantidad;
      var fechaGasto = datos.Fecha || datos.fecha || formatDateNative(new Date());
      var detalleBase = datos.Detalle || datos.detalle || "";

      var ciclosData = [];
      var ciclosSheet = ss.getSheetByName(HOJA_CICLOS);
      if (ciclosSheet && ciclosSheet.getLastRow() >= 2) {
        ciclosData = leerTablaHoja(HOJA_CICLOS);
      }

      var headersInSheet = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];

      lotesActivos.forEach(function(loteAct, idx) {

          var cLote = String(c.Lote || c.lote || c.Lote_Principal || '').trim();
          return (cLote === loteAct || cLote.indexOf(loteAct + " ") === 0) && String(c.Estado || c.estado || '').toUpperCase() === 'EN PROCESO';
        });

        var loteTrackCode = "";
        var loteGenCode = "";
        var loteCultivo = "";

        if (activeCycle) {
          loteTrackCode = activeCycle.Codigo_Desarrollo || activeCycle.codigo_desarrollo || activeCycle.Codigo_Seguimiento || activeCycle.codigo_seguimiento || activeCycle.Codigo_General || activeCycle.codigo_general || "";
          loteGenCode = activeCycle.Codigo_General || activeCycle.codigo_general || loteTrackCode;
          loteCultivo = activeCycle.Cultivo || activeCycle.cultivo || "";
        }

        if (!loteTrackCode) {
          loteTrackCode = generarCodigoSeguimiento(loteAct, tipoInsumo, fechaGasto);
          loteGenCode = loteTrackCode;
        }

        var rowValues = [];
        for (var c = 0; c < headersInSheet.length; c++) {
          var h = headersInSheet[c].toString().trim();
          var val = "";

          if (h === "Lote_Principal") val = loteAct;
          else if (h === "Sub_Lote" || h === "Division") val = "";
          else if (h === "Lote" || h === "Lote_Completo") val = loteAct;
          else if (h === "Tipo_Insumo") val = tipoInsumo;
          else if (h === "Cantidad") val = propCantidad;

          else if (h === "Estado_Pago") val = estadoPago;
          else if (h === "Cliente_Deudor_Proveedor") val = cliente;
          else if (h === "Fecha_Vencimiento") val = fechaVenc;
          else if (h === "Fecha_Pago_Realizado") val = fechaPagoReal;
          else if (h === "Fecha") val = fechaGasto;
          else if (h === "Detalle") val = (detalleBase ? detalleBase + " " : "") + "(Prorrateo general " + (idx + 1) + "/" + numLotes + " entre lotes activos)";

      });

      return { success: true, prorrateado: true, numLotes: numLotes, totalDistribuido: total };
    }

    var lotePrincipal = datos.Lote_Principal || datos.lote_principal || datos.Lote || datos.lote || "";
    var matchMain = lotePrincipal ? lotePrincipal.match(/^(Lote\s*\d+)/i) : null;
    if (matchMain) {
      lotePrincipal = matchMain[1];
    }
    var subLote = "";
    var loteCompleto = lotePrincipal;

    var cultivoItem = datos.Cultivo_Semilla || datos.cultivo_semilla || datos.Producto_Vendido || datos.producto_vendido || datos.Cultivo || datos.cultivo || "";
    var trackingCode = datos.Codigo_Seguimiento || datos.codigo_seguimiento || datos.Codigo_General || datos.codigo_general || "";
    if (!trackingCode && loteCompleto) {
      trackingCode = generarCodigoSeguimiento(loteCompleto, cultivoItem, datos.Fecha || datos.fecha);
    }
    var codigoGeneral = datos.Codigo_General || datos.codigo_general || trackingCode;

    var headersInSheet = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    var rowValues = [];

    for (var c = 0; c < headersInSheet.length; c++) {
      var h = headersInSheet[c].toString().trim();

      if (h === "Lote_Principal") val = lotePrincipal;
      else if (h === "Sub_Lote" || h === "Division") val = subLote;

      else if (h === "Cultivo_Semilla") val = datos.Cultivo_Semilla || datos.cultivo_semilla || "";

    }

  } catch (err) {

function actualizarEstadoPago(tipoTabla, filaId, nuevoEstado, fechaPago) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "";
    if (tipoTabla === "gastosIniciales" || tipoTabla === "inicial") sheetName = HOJA_GASTOS_INICIALES;
    else if (tipoTabla === "gastosDesarrollo" || tipoTabla === "desarrollo") sheetName = HOJA_GASTOS_DESARROLLO;
    else if (tipoTabla === "ventas" || tipoTabla === "venta") sheetName = HOJA_VENTAS;
    else return { success: false, error: "Tipo de tabla no válido." };

    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) return { success: false, error: "La hoja no existe." };

    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { success: false, error: "No hay registros." };

    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];

    var estadoCol = -1;
    var fechaPagoCol = -1;
    var trackingCol = -1;

    for (var c = 0; c < headers.length; c++) {
      var h = headers[c].toString().trim();
      if (h === "Estado_Pago") estadoCol = c + 1;
      if (h === "Fecha_Pago_Realizado") fechaPagoCol = c + 1;
      if (h === "Codigo_Seguimiento" || h === "Codigo_General") trackingCol = c + 1;
    }

    var rowToUpdate = -1;
    if (trackingCol > 0) {
      var trackingValues = sheet.getRange(2, trackingCol, lastRow - 1, 1).getValues();
      for (var r = 0; r < trackingValues.length; r++) {
        if (trackingValues[r][0].toString().trim().toUpperCase() === filaId.toString().trim().toUpperCase()) {
          rowToUpdate = r + 2;
          break;
        }
      }
    }

    if (rowToUpdate === -1 && Number(filaId) >= 2 && Number(filaId) <= lastRow) {
      rowToUpdate = Number(filaId);
    }

    if (rowToUpdate !== -1) {
      if (estadoCol > 0) sheet.getRange(rowToUpdate, estadoCol).setValue(nuevoEstado.toUpperCase());
      if (fechaPagoCol > 0) sheet.getRange(rowToUpdate, fechaPagoCol).setValue(fechaPago || formatDateNative(new Date()));
      return { success: true };
    }

    return { success: false, error: "No se encontró la fila especificada." };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function leerTablaHoja(sheetName) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return getSheetData(ss, sheetName);
}

function getSheetData(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];

  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];

  var lastCol = sheet.getLastColumn();
  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

  var result = [];
  for (var r = 0; r < values.length; r++) {
    var rowData = {};
    var hasData = false;
    for (var c = 0; c < headers.length; c++) {
      var headerName = headers[c].toString().trim();
      if (!headerName) continue;
      var cellVal = values[r][c];
      if (cellVal instanceof Date) {
        cellVal = formatDateNative(cellVal);
      }
      if (cellVal !== "" && cellVal !== null && cellVal !== undefined) {
        hasData = true;
      }
      rowData[headerName] = cellVal;
    }
    if (hasData) {
      result.push(rowData);
    }
  }
  return result;
}

function agregarCiclo(lotePrincipal, subLote, cultivo, codigoGeneral, codigoInicial, codigoDesarrollo, codigoVentas, estado, fechaInicio, fechaFin) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(HOJA_CICLOS);
    var columnsNeeded = ["Lote_Principal", "Sub_Lote", "Lote", "Cultivo", "Codigo_General", "Codigo_Inicial", "Codigo_Desarrollo", "Codigo_Ventas", "Estado", "Fecha_Inicio", "Fecha_Fin"];
    if (!sheet) {
      sheet = ss.insertSheet(HOJA_CICLOS);
      sheet.appendRow(columnsNeeded);
    }
    asegurarColumnas(sheet, columnsNeeded);

    var lp = lotePrincipal || "";
    var sub = subLote || "";
    var cult = cultivo || "";
    var cg = codigoGeneral || "";
    var ci = codigoInicial || "";
    var cd = codigoDesarrollo || "";
    var cv = codigoVentas || "";
    var est = estado || "EN PROCESO";
    var fi = fechaInicio || formatDateNative(new Date());
    var ff = fechaFin || "";

    var loteComp = "";
    if (typeof lotePrincipal === 'object' && lotePrincipal !== null) {
      var obj = lotePrincipal;
      lp = obj.Lote_Principal || obj.lote_principal || obj.Lote || obj.lote || "";
      sub = "";
      loteComp = lp;
      cult = obj.Cultivo || obj.cultivo || "";
      cg = obj.Codigo_General || obj.codigo_general || "";
      ci = obj.Codigo_Inicial || obj.codigo_inicial || "";
      cd = obj.Codigo_Desarrollo || obj.codigo_desarrollo || "";
      cv = obj.Codigo_Ventas || obj.codigo_ventas || "";
      est = obj.Estado || obj.estado || "EN PROCESO";
      fi = obj.Fecha_Inicio || obj.fecha_inicio || formatDateNative(new Date());
      ff = obj.Fecha_Fin || obj.fecha_fin || "";
    } else {
      sub = "";
      var matchL = lp ? lp.match(/^(Lote\s*\d+)/i) : null;
      if (matchL) {
        lp = matchL[1];
      }
      loteComp = lp;
    }

    if (!cg) cg = generarCodigoSeguimiento(loteComp, cult, fi);
    if (!ci) ci = cg;
    if (!cd) cd = cg;
    if (!cv) cv = cg;

    var rowValues = [lp, sub, loteComp, cult, cg, ci, cd, cv, est, fi, ff];
    sheet.appendRow(rowValues);
    return { success: true, codigoGeneral: cg };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function finalizarCiclo(codigoGeneral, fechaFin) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(HOJA_CICLOS);
    if (!sheet) return { success: false, error: "No existe la hoja de ciclos." };

    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { success: false, error: "No hay ciclos registrados." };

    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var cgCol = -1, estCol = -1, ffCol = -1;

    for (var c = 0; c < headers.length; c++) {
      var h = headers[c].toString().trim();
      if (h === "Codigo_General" || h === "Codigo_Inicial" || h === "Codigo_Seguimiento") cgCol = c + 1;
      if (h === "Estado") estCol = c + 1;
      if (h === "Fecha_Fin") ffCol = c + 1;
    }

    if (cgCol === -1) cgCol = 5;
    if (estCol === -1) estCol = 9;
    if (ffCol === -1) ffCol = 11;

    var range = sheet.getRange(2, 1, lastRow - 1, lastCol);
    var values = range.getValues();

    var targetUpper = codigoGeneral ? codigoGeneral.toString().trim().toUpperCase() : "";

    for (var i = 0; i < values.length; i++) {
      var row = values[i];
      var match = false;

      for (var colIdx = 0; colIdx < row.length; colIdx++) {
        var cellStr = row[colIdx] ? row[colIdx].toString().trim().toUpperCase() : "";
        if (targetUpper && cellStr === targetUpper) {
          match = true;
          break;
        }
      }

      if (match || (targetUpper === "" && row[estCol - 1].toString().trim().toUpperCase() === "EN PROCESO")) {
        sheet.getRange(i + 2, estCol).setValue("FINALIZADO");
        sheet.getRange(i + 2, ffCol).setValue(fechaFin || formatDateNative(new Date()));
        break;
      }
    }
    return { success: true };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function asegurarColumnas(sheet, headersNeeded) {
  var lastCol = sheet.getLastColumn();
  if (lastCol === 0) {
    sheet.appendRow(headersNeeded);
    return;
  }
  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  var headerMap = {};
  for (var i = 0; i < headers.length; i++) {
    headerMap[headers[i].toString().trim()] = true;
  }
  for (var j = 0; j < headersNeeded.length; j++) {
    var h = headersNeeded[j];
    if (!headerMap[h]) {
      sheet.getRange(1, sheet.getLastColumn() + 1).setValue(h);
    }
  }
}

function formatDateNative(date) {
  var d = new Date(date),
      month = '' + (d.getMonth() + 1),
      day = '' + d.getDate(),
      year = d.getFullYear();

  if (month.length < 2) month = '0' + month;
  if (day.length < 2) day = '0' + day;

  return [year, month, day].join('-');
}

function obtenerBalanceCajaGeneral(filtros) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = {
      gastosIniciales: getSheetData(ss, HOJA_GASTOS_INICIALES),
      gastosDesarrollo: getSheetData(ss, HOJA_GASTOS_DESARROLLO),
      ventas: getSheetData(ss, HOJA_VENTAS),
      gastosPersonales: getSheetData(ss, HOJA_GASTOS_PERSONALES),
      ingresosExternos: getSheetData(ss, HOJA_INGRESOS_EXTERNOS)
    };

    var fechaInicio = filtros ? filtros.fechaInicio : null;
    var fechaFin = filtros ? filtros.fechaFin : null;

    var isDateOk = function(dateStr) {
      if (!dateStr) return true;
      if (fechaInicio && dateStr < fechaInicio) return false;
      if (fechaFin && dateStr > fechaFin) return false;
      return true;
    };

    var ventasCobradas = data.ventas
      .filter(function(x) { return isDateOk(x.Fecha || x.fecha) && String(x.Estado_Pago || x.estado_pago || 'PAGADO').toUpperCase() === 'PAGADO'; })
      .reduce(function(sum, x) { return sum + (Number(x.Ingreso_Total_COP || x.ingreso_total_cop) || 0); }, 0);

    var gastosAgricolasPagados = data.gastosIniciales.concat(data.gastosDesarrollo)
      .filter(function(x) { return isDateOk(x.Fecha || x.fecha) && String(x.Estado_Pago || x.estado_pago || 'PAGADO').toUpperCase() === 'PAGADO'; })
      .reduce(function(sum, x) { return sum + (Number(x.Costo_Total_COP || x.costo_total_cop) || 0); }, 0);

    var cajaRealAgricola = ventasCobradas - gastosAgricolasPagados;

    var ingresosExternosTotales = data.ingresosExternos
      .filter(function(x) { return isDateOk(x.Fecha || x.fecha); })
      .reduce(function(sum, x) { return sum + (Number(x.Monto_COP || x.monto_cop || x.Monto || x.monto) || 0); }, 0);

    var gastosPersonalesTotales = data.gastosPersonales
      .filter(function(x) { return isDateOk(x.Fecha || x.fecha); })
      .reduce(function(sum, x) { return sum + (Number(x.Monto_COP || x.monto_cop) || 0); }, 0);

    var saldoLiquidoDisponible = (cajaRealAgricola + ingresosExternosTotales) - gastosPersonalesTotales;

    return {
      success: true,
      metrics: {
        cajaRealAgricola: cajaRealAgricola,
        ingresosExternosTotales: ingresosExternosTotales,
        gastosPersonalesTotales: gastosPersonalesTotales,
        saldoLiquidoDisponible: saldoLiquidoDisponible,
        ventasCobradas: ventasCobradas,
        gastosAgricolasPagados: gastosAgricolasPagados
      }
    };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function agregarItemConfig(columna, nuevoValor) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(HOJA_CONFIGURACION);
    if (!sheet) {
      return { success: false, error: "La hoja de configuración no existe." };
    }

    var lastRow = sheet.getLastRow();
    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];

    var colIndex = -1;
    for (var i = 0; i < headers.length; i++) {
      if (headers[i].toString().trim() === columna) {
        colIndex = i + 1;
        break;
      }
    }

    if (colIndex === -1) {
      colIndex = lastCol + 1;
      sheet.getRange(1, colIndex).setValue(columna);
    }

    var targetRow = 2;
    if (lastRow >= 2) {
      var colValues = sheet.getRange(2, colIndex, lastRow - 1, 1).getValues();
      for (var r = 0; r < colValues.length; r++) {
        if (colValues[r][0].toString().trim() === "") {
          targetRow = r + 2;
          break;
        }
        targetRow = r + 3;
      }
    }

    sheet.getRange(targetRow, colIndex).setValue(nuevoValor);
    return { success: true };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function eliminarRegistroSeguro(tipoTabla, idRegistro, codigoSeguridad) {
  try {
    var creds = getSecurityCredentials();
    if (codigoSeguridad.toString().trim() !== creds.securityCode.toString().trim()) {
      return { success: false, message: "Código de seguridad incorrecto. Operación cancelada." };
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "";
    if (tipoTabla === "gastosIniciales" || tipoTabla === "inicial") sheetName = HOJA_GASTOS_INICIALES;
    else if (tipoTabla === "gastosDesarrollo" || tipoTabla === "desarrollo") sheetName = HOJA_GASTOS_DESARROLLO;
    else if (tipoTabla === "ventas" || tipoTabla === "venta") sheetName = HOJA_VENTAS;
    else if (tipoTabla === "gastosPersonales" || tipoTabla === "personal") sheetName = HOJA_GASTOS_PERSONALES;
    else if (tipoTabla === "fotosLotes" || tipoTabla === "foto") sheetName = HOJA_FOTOS_LOTES;
    else if (tipoTabla === "ingresosExternos" || tipoTabla === "ingresoExterno") sheetName = HOJA_INGRESOS_EXTERNOS;
    else if (tipoTabla === "metasRecoleccion" || tipoTabla === "metaRecoleccion") sheetName = HOJA_METAS_RECOLECCION;
    else if (tipoTabla === "aportesMetas" || tipoTabla === "aporteMeta") sheetName = HOJA_APORTES_METAS;
    else return { success: false, message: "Tipo de tabla no válido." };

    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) return { success: false, message: "La hoja no existe." };

    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { success: false, message: "No hay registros en la hoja." };

    var lastCol = sheet.getLastColumn();
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];

    var codeColIdx = -1;
    for (var c = 0; c < headers.length; c++) {
      var h = headers[c].toString().trim();
      if (h === "Codigo_Seguimiento" || h === "Codigo_General" || h === "ID_Gasto_Personal" || h === "ID_Foto" || h === "ID_Ingreso_Externo" || h === "ID_Meta" || h === "ID_Aporte") {
        codeColIdx = c + 1;
        break;
      }
    }

    var rowToDelete = -1;
    var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

    for (var r = 0; r < values.length; r++) {
      var row = values[r];
      var match = false;

      if (codeColIdx > 0 && row[codeColIdx - 1].toString().trim().toUpperCase() === idRegistro.toString().trim().toUpperCase()) {
        match = true;
      } else if (r + 2 === Number(idRegistro)) {
        match = true;
      }

      if (!match) {
        for (var k = 0; k < row.length; k++) {
          if (row[k].toString().trim().toUpperCase() === idRegistro.toString().trim().toUpperCase()) {
            match = true;
            break;
          }
        }
      }

      if (match) {
        rowToDelete = r + 2;
        break;
      }
    }

    if (rowToDelete === -1 && Number(idRegistro) >= 2 && Number(idRegistro) <= lastRow) {
      rowToDelete = Number(idRegistro);
    }

    if (rowToDelete !== -1) {
      var trackingCodeFound = "";
      if (codeColIdx > 0 && values[rowToDelete - 2]) {
        trackingCodeFound = values[rowToDelete - 2][codeColIdx - 1].toString().trim().toUpperCase();
      }
      if (!trackingCodeFound) trackingCodeFound = idRegistro.toString().trim().toUpperCase();

      var loteFound = "";
      var loteColIdx = -1;
      for (var lc = 0; lc < headers.length; lc++) {
        var hName = headers[lc].toString().trim();
        if (hName === "Lote" || hName === "Lote_Principal") { loteColIdx = lc; break; }
      }
      if (loteColIdx >= 0 && values[rowToDelete - 2]) {
        loteFound = values[rowToDelete - 2][loteColIdx].toString().trim();
      }

      sheet.deleteRow(rowToDelete);

      // Cascaded deletion: check and sync cycles sheet
      if (trackingCodeFound || loteFound) {
        sincronizarEliminacionCiclo(ss, trackingCodeFound, loteFound);
      }

      return { success: true, idEliminado: idRegistro, tipoTabla: tipoTabla, message: "Registro eliminado correctamente." };
    }

    return { success: false, message: "No se encontró el registro especificado." };
  } catch (err) {
    return { success: false, message: err.toString() };
  }
}

function sincronizarEliminacionCiclo(ss, trackingCode, loteName) {
  try {
    var codeUpper = (trackingCode || "").toString().trim().toUpperCase();
    var loteClean = (loteName || "").toString().trim().toUpperCase();

    // Check if any transaction remains in gastosIniciales, gastosDesarrollo or ventas for this trackingCode or Lote
    var gIni = getSheetData(ss, HOJA_GASTOS_INICIALES);
    var gDes = getSheetData(ss, HOJA_GASTOS_DESARROLLO);
    var vts = getSheetData(ss, HOJA_VENTAS);

    var hasRemainingTxForCode = function(arr) {
      return (arr || []).some(function(item) {
        var c = (item.Codigo_Seguimiento || item.codigo_seguimiento || item.Codigo_General || item.codigo_general || "").toString().trim().toUpperCase();
        return codeUpper && c === codeUpper;
      });
    };

    var hasRemainingTxForLote = function(arr) {
      return (arr || []).some(function(item) {
        var l = (item.Lote || item.lote || item.Lote_Principal || item.lote_principal || "").toString().trim().toUpperCase();
        return loteClean && (l === loteClean || l.indexOf(loteClean) === 0);
      });
    };

    var codeHasRecords = hasRemainingTxForCode(gIni) || hasRemainingTxForCode(gDes) || hasRemainingTxForCode(vts);
    var loteHasRecords = hasRemainingTxForLote(gIni) || hasRemainingTxForLote(gDes) || hasRemainingTxForLote(vts);

    var ciclosSheet = ss.getSheetByName(HOJA_CICLOS);
    if (!ciclosSheet) return;

    var lastRow = ciclosSheet.getLastRow();
    if (lastRow < 2) return;

    var lastCol = ciclosSheet.getLastColumn();
    var headers = ciclosSheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var values = ciclosSheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

    // Delete rows in cycles sheet if no records remain for the code or for the lot
    for (var i = values.length - 1; i >= 0; i--) {
      var row = values[i];
      var rowCode = "";
      var rowLote = "";

      for (var c = 0; c < headers.length; c++) {
        var hName = headers[c].toString().trim();
        if (hName === "Codigo_General" || hName === "Codigo_Inicial" || hName === "Codigo_Seguimiento") {
          rowCode = (row[c] || "").toString().trim().toUpperCase();
        }
        if (hName === "Lote" || hName === "Lote_Principal") {
          rowLote = (row[c] || "").toString().trim().toUpperCase();
        }
      }

      var matchCode = codeUpper && rowCode === codeUpper;
      var matchLote = loteClean && (rowLote === loteClean || rowLote.indexOf(loteClean) === 0);

      if ((matchCode && !codeHasRecords) || (matchLote && !loteHasRecords)) {
        ciclosSheet.deleteRow(i + 2);
      }
    }
  } catch (err) {
    Logger.log("Error en sincronizarEliminacionCiclo: " + err.toString());
  }
}
